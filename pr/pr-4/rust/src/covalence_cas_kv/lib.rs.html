1//! A content-addressed store over a key-value store.
2//!
3//! [`KvCas`] keeps blocks in any [`Kv`], such as `covalence-kv-memory`'s
4//! `MemoryKv` or `covalence-kv-s3`'s `S3Kv`. Each block lies under
5//! `blake3/<digest>/bytes`, after an optional prefix, where `<digest>` is its
6//! [BLAKE3](https://github.com/BLAKE3-team/BLAKE3-specs) digest in lowercase,
7//! unpadded [base32](https://www.rfc-editor.org/rfc/rfc4648#section-6).
8//!
9//! ```
10//! use covalence_cas_kv::KvCas;
11//! use covalence_cas_trait::{GetExt, Put};
12//! use covalence_kv_memory::MemoryKv;
13//! use covalence_kv_trait::Kv;
14//! use covalence_lib_cid::Cid;
15//! let kv = MemoryKv::default().into_dyn();
16//! let store = KvCas::with_prefix(kv.clone(), "cas")?;
17//! let cid: Cid = store.put("hello".into())?;
18//! assert_eq!(store.get_fact(&cid)?.unwrap().bytes(), "hello");
19//! // BLAKE3("hello") is ea8f163d...4a67200f in hex.
20//! let key = "cas/blake3/5khrmpntq2bjexseshc6ldklwnig56gbj23yvbxjbdcwestheahq/bytes";
21//! assert_eq!(kv.get(key.as_bytes())?.unwrap(), "hello");
22//! # Ok::<(), Box<dyn std::error::Error>>(())
23//! ```
24#![deny(missing_docs)]
25

41/// A store of blocks in a [`Kv`]; see the [crate] docs for the key layout.
42///
43/// - A CID with a 32-byte BLAKE3 digest finds its bytes whatever its codec;
44///   other CIDs find nothing. [`Put`] returns the raw BLAKE3 CID.
45/// - The key-value store may be remote or shared, so `get`, `aget`,
46///   `get_fact` and `aget_fact` check the bytes against the CID and fail with
47///   [`Error::Check`] if they do not match. `get_unchecked`, sizes and ranges
48///   are not checked.
49/// - `has` asks the key-value store whether the key exists (a HEAD request on
50///   S3), without reading the bytes.
51/// - `put` writes without asking first, so it needs only write access. (S3
52///   answers a HEAD for a missing key with 403 to a client that may not list
53///   the bucket.)
54/// - `put_from` and `get_to` stream through the key-value store's own
55///   `put_from` and `get_to`, hashing the bytes as they pass.
56/// - If the bytes `put_from` reads do not hash to the CID, the read that would
57///   hand over the last byte fails instead. The key-value store's `put_from`
58///   then fails as [`Kv::put_from`] says, storing nothing, and `put_from`
59///   fails with [`Error::Check`]; or with [`Error::Backend`] if the key-value
60///   store reports a further failure, such as failing to clean up.
61/// - [`Cas::info`] is `{"name": "kv", "prefix": <prefix>, "kv": <the
62///   key-value store's info>}`; [`Cas::stats`] is the key-value store's.
63pub struct KvCas {
64    kv: Arc<dyn Kv>,
65    prefix: String,
66}
67
68/// A prefix that is not `/`-separated, non-empty segments other than `.` and
69/// `..`, optionally ending in `/`.
70#[derive(Debug, Snafu)]
71#[snafu(crate_root(covalence_lib_error::snafu))]
72#[snafu(display("invalid key prefix {prefix:?}"))]
73pub struct InvalidPrefix {
74    prefix: String,
75}

77impl KvCas {
78    /// A store with keys at the root of `kv`.
79    pub fn new(kv: Arc<dyn Kv>) -> Self {
80        Self {
81            kv,
82            prefix: String::new(),
83        }
84    }
85
86    /// A store with keys under `prefix/` in `kv`; a trailing `/` on `prefix`
87    /// is optional.
88    ///
89    /// Fails with [`InvalidPrefix`] unless `prefix` is `/`-separated,
90    /// non-empty segments other than `.` and `..`, so the empty prefix fails:
91    /// use [`KvCas::new`] instead.
92    ///
93    /// ```
94    /// use covalence_cas_kv::KvCas;
95    /// use covalence_kv_memory::MemoryKv;
96    /// use covalence_kv_trait::Kv;
97    /// let kv = MemoryKv::default().into_dyn();
98    /// assert!(KvCas::with_prefix(kv.clone(), "a/b/").is_ok());
99    /// for prefix in ["", "/", "/a", "a//b", "a/../b"] {
100    ///     assert!(KvCas::with_prefix(kv.clone(), prefix).is_err());
101    /// }
102    /// ```
103    pub fn with_prefix(kv: Arc<dyn Kv>, prefix: &str) -> Result<Self, InvalidPrefix> {
104        let trimmed = prefix.strip_suffix('/').unwrap_or(prefix);
105        if trimmed.split('/').any(|s| matches!(s, "" | "." | "..")) {
106            return InvalidPrefixSnafu { prefix }.fail();
107        }
108        Ok(Self {
109            kv,
110            prefix: format!("{trimmed}/"),
111        })
112    }
113
114    /// The key for `cid`.
115    fn key(&self, cid: &Blake3) -> String {
116        let digest = BASE32_NOPAD.encode(cid.digest()).to_ascii_lowercase();
117        format!("{}blake3/{digest}/bytes", self.prefix)
118    }
119}

121fn backend(error: covalence_kv_trait::Error) -> Error {
122    match error {
123        covalence_kv_trait::Error::Io { source } => Error::Io { source },
124        error => Error::Backend {
125            source: Box::new(error),
126        },
127    }
128}

130/// Whether `hasher` has hashed exactly the bytes of `cid`.
131fn hashes_to(hasher: &blake3::Hasher, cid: &Blake3) -> bool {
132    hasher.finalize().as_bytes() == cid.digest()
133}

135/// The next `left` bytes of `reader`, which must hash to `cid`: the read
136/// returning the last of them fails instead if they do not.
137struct Checked<'a> {
138    reader: &'a mut dyn Read,
139    hasher: blake3::Hasher,
140    left: u64,
141    cid: Blake3,
142    mismatch: bool,
143}
144
145impl Read for Checked<'_> {
146    fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
147        let max = buf
148            .len()
149            .min(usize::try_from(self.left).unwrap_or(usize::MAX));
150        if max == 0 {
151            return Ok(0);
152        }
153        let read = self.reader.read(&mut buf[..max])?;
154        self.hasher.update(&buf[..read]);
155        self.left -= read as u64;
156        if self.left == 0 && !hashes_to(&self.hasher, &self.cid) {
157            self.mismatch = true;
158            let error = covalence_lib_cid::Error::Mismatch;
159            return Err(io::Error::new(io::ErrorKind::InvalidData, error));
160        }
161        Ok(read)
162    }
163}

165/// A writer hashing what it passes on.
166struct Hashing<'a> {
167    writer: &'a mut dyn Write,
168    hasher: blake3::Hasher,
169}
170
171impl Write for Hashing<'_> {
172    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
173        let written = self.writer.write(buf)?;
174        self.hasher.update(&buf[..written]);
175        Ok(written)
176    }
177
178    fn flush(&mut self) -> io::Result<()> {
179        self.writer.flush()
180    }
181}

234            let Some(key) = key.raw_blake3() else {
235                return Ok(None);

276impl Put<Blake3> for KvCas {
277    fn put(&self, bytes: Bytes) -> Result<Blake3, Error> {
278        let cid = Blake3::of(&bytes);

281            .map_err(backend)?;

303impl Cas for KvCas {
304    fn has(&self, key: &Cid) -> Result<bool, Error> {
305        key.raw_blake3().map_or(Ok(false), |key| {
306            let key = self.key(&key);
307            self.kv.exists(key.as_bytes()).map_err(backend)
308        })
309    }
310
311    /// The name `kv`, the key prefix, and the key-value store's info.
312    fn info(&self) -> Result<Value, Error> {
313        Ok(map([
314            ("name", Value::String("kv".into())),
315            ("prefix", Value::String(self.prefix.as_str().into())),
316            ("kv", self.kv.info().map_err(backend)?),
317        ]))
318    }
319
320    fn stats(&self) -> Result<Value, Error> {
321        self.kv.stats().map_err(backend)
322    }
323
324    fn put_from(&self, cid: Blake3, reader: &mut dyn Read, len: u64) -> Result<Cid, Error> {
325        // No read hands over the last byte of nothing, so check it here.
326        if len == 0 && !hashes_to(&blake3::Hasher::new(), &cid) {
327            return Err(covalence_lib_cid::Error::Mismatch.into());
328        }
329        let mut checked = Checked {
330            reader,
331            hasher: blake3::Hasher::new(),
332            left: len,
333            cid,
334            mismatch: false,
335        };
336        let key = self.key(&cid);
337        match self.kv.put_from(key.as_bytes(), &mut checked, len) {
338            Ok(()) => Ok(cid.into()),
339            // The store failed as asked; a store error beyond that, such as a
340            // failed cleanup, is the one to report.
341            Err(covalence_kv_trait::Error::Io { .. }) if checked.mismatch => {
342                Err(covalence_lib_cid::Error::Mismatch.into())
343            }
344            Err(error) => Err(backend(error)),
345        }
346    }
347
348    fn get_to(&self, key: &Cid, writer: &mut dyn Write) -> Result<Option<u64>, Error> {
349        let Some(cid) = key.raw_blake3() else {
350            return Ok(None);
351        };
352        let mut hashing = Hashing {
353            writer,
354            hasher: blake3::Hasher::new(),
355        };
356        let key = self.key(&cid);
357        let Some(len) = self
358            .kv
359            .get_to(key.as_bytes(), &mut hashing)
360            .map_err(backend)?
361        else {
362            return Ok(None);
363        };
364        if !hashes_to(&hashing.hasher, &cid) {
365            return Err(covalence_lib_cid::Error::Mismatch.into());
366        }
367        Ok(Some(len))
368    }
