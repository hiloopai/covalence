17/// The header carrying the base64 SHA-256 of a PUT body, signed with the URL.
18const CHECKSUM: &str = "x-amz-checksum-sha256";
19
20/// An uploaded part: its ETag and base64 SHA-256.
21struct Part {
22    etag: String,
23    checksum: String,
24}

26/// `text` escaped for an XML element.
27fn escape(text: &str) -> String {
28    text.replace('&', "&amp;").replace('<', "&lt;")
29}

31impl S3Kv {
32    /// Store `value` under `key`.
33    pub(crate) fn write(&self, key: &[u8], value: Bytes) -> Result<(), Error> {
34        let mut offset = 0;
35        self.upload(key, value.len() as u64, |len| {
36            let part = value.slice(offset..offset + len);
37            offset += len;
38            Ok(part)
39        })
40    }
41
42    /// Store the next `len` bytes of `reader` under `key`.
43    pub(crate) fn write_from(
44        &self,
45        key: &[u8],
46        reader: &mut dyn Read,
47        len: u64,
48    ) -> Result<(), Error> {
49        self.upload(key, len, |len| {
50            let mut part = vec![0; len];
51            reader.read_exact(&mut part).context(IoSnafu)?;
52            Ok(part.into())
53        })
54    }
55
56    /// Store `len` bytes under `key`, taking each part's bytes from `next`
57    /// in order, which is asked for all of them at once if they fit in one
58    /// part. Parts go up concurrently; the upload is aborted only once none
59    /// is in flight, so that no part outlives it.
60    fn upload(
61        &self,
62        key: &[u8],
63        len: u64,
64        mut next: impl FnMut(usize) -> Result<Bytes, Error>,
65    ) -> Result<(), Error> {
66        let object = self.object(key)?;
67        let part = self.part_for(len);
68        if len <= part {
69            return self.put_object(&object, next(len as usize)?);
70        }
71        let upload_id = self.create(&object)?;
72        // At most MAX_PARTS, so the count and part numbers fit.
73        let count = len.div_ceil(part);
74        let mut parts = Vec::with_capacity(count as usize);
75        let jobs = (0..count).map(|index| {
76            let size = part.min(len - index * part);
77            Ok((index as u16 + 1, next(size as usize)?))
78        });
79        // Part ETags are small, so only the parts in flight bound memory, and
80        // one slow part does not hold up the rest.
81        let result = self
82            .in_order(
83                count as usize,
84                count as usize,
85                jobs,
86                |(number, bytes)| self.upload_part(&object, &upload_id, number, &bytes),
87                |uploaded| {
88                    parts.push(uploaded);
89                    Ok(())
90                },
91            )
92            .and_then(|()| self.complete(&object, &upload_id, &parts));
93        result.map_err(|cause| match self.abort(&object, &upload_id) {
94            Ok(()) => cause,
95            Err(source) => Error::Abort {
96                upload_id,
97                cause: Box::new(cause),
98                source: Box::new(source),
99            },
100        })
101    }
102
103    /// The part size for a value of `len` bytes: the configured one, or as
104    /// many bytes as keep it to [`MAX_PARTS`] parts.
105    fn part_for(&self, len: u64) -> u64 {
106        self.part.max(len.div_ceil(MAX_PARTS))
107    }
108
109    /// PUT `value` as `object` in one request.
110    fn put_object(&self, object: &str, value: Bytes) -> Result<(), Error> {
111        let checksum = BASE64.encode(&Sha256::digest(&value));
112        let credentials = self.credentials()?;
113        let (status, body) = self.send(|| {
114            let mut action = self.bucket.put_object(Some(&credentials), object);
115            action.headers_mut().insert(CHECKSUM, &checksum);
116            let url = action.sign(EXPIRY);
117            let request = self.agent.put(url.as_str()).config();
118            let request = request.timeout_global(Some(timeout(value.len() as u64)));
119            let request = request.build().header(CHECKSUM, &checksum);
120            body(request.send(&value[..])?, MAX_ERROR, Vec::new())
121        })?;
122        match status {
123            200 => Ok(()),
124            status => status_error(status, &body),
125        }
126    }
127
128    /// Start a multipart upload of `object` with SHA-256 part checksums, and
129    /// return its ID.
130    fn create(&self, object: &str) -> Result<String, Error> {
131        const ALGORITHM: &str = "x-amz-checksum-algorithm";
132        let credentials = self.credentials()?;
133        let (status, body) = self.send(|| {
134            let mut action = self
135                .bucket
136                .create_multipart_upload(Some(&credentials), object);
137            action.headers_mut().insert(ALGORITHM, "SHA256");
138            let url = action.sign(EXPIRY);
139            let request = self.agent.post(url.as_str()).header(ALGORITHM, "SHA256");
140            body(request.send_empty()?, MAX_ERROR, Vec::new())
141        })?;
142        if status != 200 {
143            return status_error(status, &body);
144        }
145        let id = str::from_utf8(&body)
146            .ok()
147            .and_then(|body| CreateMultipartUpload::parse_response(body).ok())
148            .context(MissingFieldSnafu { field: "upload ID" })?;
149        Ok(id.upload_id().to_owned())
150    }
151
152    /// Upload `bytes` as part `number` of upload `upload_id`.
153    fn upload_part(
154        &self,
155        object: &str,
156        upload_id: &str,
157        number: u16,
158        bytes: &[u8],
159    ) -> Result<Part, Error> {
160        let checksum = BASE64.encode(&Sha256::digest(bytes));
161        let credentials = self.credentials()?;
162        let (status, (body, etag)) = self.send(|| {
163            let mut action = self
164                .bucket
165                .upload_part(Some(&credentials), object, number, upload_id);
166            action.headers_mut().insert(CHECKSUM, &checksum);
167            let url = action.sign(EXPIRY);
168            let request = self.agent.put(url.as_str()).config();
169            let request = request.timeout_global(Some(timeout(bytes.len() as u64)));
170            let request = request.build().header(CHECKSUM, &checksum);
171            let response = request.send(bytes)?;
172            let etag = response
173                .headers()
174                .get("etag")
175                .and_then(|etag| etag.to_str().ok())
176                .map(str::to_owned);
177            let (status, body) = body(response, MAX_ERROR, Vec::new())?;
178            Ok((status, (body, etag)))
179        })?;
180        if status != 200 {
181            return status_error(status, &body);
182        }
183        let etag = etag.context(MissingFieldSnafu { field: "ETag" })?;
184        Ok(Part { etag, checksum })
185    }

187    /// Complete upload `upload_id` from `parts`. S3 can answer 200 with an
188    /// error in the body, so that is a failure too.
189    fn complete(&self, object: &str, upload_id: &str, parts: &[Part]) -> Result<(), Error> {
190        let mut xml = String::from("<CompleteMultipartUpload>");
191        for (number, part) in (1..).zip(parts) {
192            xml += &format!(
193                "<Part><ETag>{}</ETag><ChecksumSHA256>{}</ChecksumSHA256>\
194                 <PartNumber>{number}</PartNumber></Part>",
195                escape(&part.etag),
196                escape(&part.checksum),
197            );
198        }
199        xml += "</CompleteMultipartUpload>";
200        let credentials = self.credentials()?;
201        let (status, body) = self.send(|| {
202            // rusty-s3 writes a body without checksums, so only sign with it.
203            let action = self.bucket.complete_multipart_upload(
204                Some(&credentials),
205                object,
206                upload_id,
207                std::iter::empty::<&str>(),
208            );
209            let url = action.sign(EXPIRY);
210            // S3 takes longer to join more parts.
211            let wait = timeout(parts.len() as u64 * crate::MIN_RATE);
212            let request = self.agent.post(url.as_str()).config();
213            let request = request.timeout_global(Some(wait)).build();
214            body(request.send(&xml)?, MAX_ERROR, Vec::new())
215        })?;
216        match status {
217            200 if !body.windows(7).any(|window| window == b"<Error>") => Ok(()),
218            status => status_error(status, &body),
219        }
220    }
221
222    /// Abort upload `upload_id`, freeing its parts. A missing upload is
223    /// already gone.
224    fn abort(&self, object: &str, upload_id: &str) -> Result<(), Error> {
225        let credentials = self.credentials()?;
226        let (status, body) = self.send(|| {
227            let url = self
228                .bucket
229                .abort_multipart_upload(Some(&credentials), object, upload_id)
230                .sign(EXPIRY);
231            body(
232                self.agent.delete(url.as_str()).call()?,
233                MAX_ERROR,
234                Vec::new(),
235            )
236        })?;
237        match status {
238            200 | 204 | 404 => Ok(()),
239            status => status_error(status, &body),
240        }
241    }
242}

244#[cfg(test)]
245mod tests {
246    use crate::{Credentials, MAX_OBJECT, MAX_PARTS, PART, S3Kv};
247
248    #[test]
249    fn parts_keep_to_ten_thousand() {
250        let location = "s3://test".parse().unwrap();
251        let credentials = Credentials::new("key", "secret");
252        let kv = S3Kv::new("http://localhost", "us-east-1", location, credentials).unwrap();
253        for len in [
254            0,
255            1,
256            PART,
257            PART * MAX_PARTS,
258            PART * MAX_PARTS + 1,
259            MAX_OBJECT,
260        ] {
261            let part = kv.part_for(len);
262            assert!(part >= PART, "{len}");
263            assert!(len.div_ceil(part) <= MAX_PARTS, "{len}");
264        }
265        assert_eq!(kv.part_for(PART * MAX_PARTS), PART);
266        assert_eq!(kv.part_for(PART * MAX_PARTS + 1), PART + 1);
267    }
