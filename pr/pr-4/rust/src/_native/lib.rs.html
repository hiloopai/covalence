39/// A code that fits in a CID.
40fn code(code: u64) -> PyResult<Multicodec> {
41    if code < 1 << 63 {
42        Ok(Multicodec(code))
43    } else {
44        Err(fail(format!("code {code:#x} is too large")))
45    }
46}

48/// Defines a Python class wrapping a [`Multicodec`], named by `$names`, the
49/// first of each being the canonical name.
50macro_rules! code_class {
51    ($(#[$doc:meta])* $rust:ident, $py:literal, $arg:ident, $kind:literal, { $($code:expr => [$($name:literal),+]),+ $(,)? }) => {
52        $(#[$doc])*
53        #[pyclass(skip_from_py_object, frozen, eq, hash, name = $py, module = "covalence")]
54        #[pyo3(crate = "covalence_lib_python::pyo3")]
55        #[derive(Clone, Copy, PartialEq, Eq, Hash)]
56        struct $rust(Multicodec);
57
58        impl $rust {
59            fn named(name: &str) -> Option<Self> {
60                match name {
61                    $($($name)|+ => Some(Self($code)),)+
62                    _ => None,
63                }
64            }
65
66            fn name(&self) -> Option<&'static str> {
67                match self.0 {
68                    $(code if code == $code => Some([$($name),+][0]),)+
69                    _ => None,
70                }
71            }
72        }
73
74        #[pymethods]
75        #[pyo3(crate = "covalence_lib_python::pyo3")]
76        impl $rust {
77            #[new]
78            fn new(name: &str) -> PyResult<Self> {
79                Self::named(name).ok_or_else(|| fail(format!(concat!("unknown ", $kind, " {:?}"), name)))
80            }
81
82            #[doc = concat!("The ", $kind, " with multicodec code `code`, named or not.\n\n- A code from 2**63 to 2**64 - 1 raises ValueError.\n- A negative code, or one of 2**64 or more, raises OverflowError.")]
83            #[staticmethod]
84            fn from_code(code: u64) -> PyResult<Self> {
85                Ok(Self(crate::code(code)?))
86            }
87
88            #[doc = concat!("The ", $kind, "'s name, or None if Covalence has none for it.")]
89            #[getter(name)]
90            fn py_name(&self) -> Option<&'static str> {
91                self.name()
92            }
93
94            /// The multicodec code.
95            #[getter]
96            fn code(&self) -> u64 {
97                self.0 .0
98            }
99
100            fn __str__(&self) -> String {
101                self.name().map_or_else(|| format!("{:#x}", self.0 .0), str::to_owned)
102            }
103
104            fn __repr__(&self) -> String {
105                match self.name() {
106                    Some(name) => format!(concat!($py, "('{}')"), name),
107                    None => format!(concat!($py, ".from_code({:#x})"), self.0 .0),
108                }
109            }
110        }
111
112        /// A `str` name or the class itself.
113        #[derive(FromPyObject)]
114        #[pyo3(crate = "covalence_lib_python::pyo3")]
115        enum $arg<'py> {
116            Object(PyRef<'py, $rust>),
117            Name(PyBackedStr),
118        }
119
120        impl $arg<'_> {
121            fn code(&self) -> PyResult<Multicodec> {
122                match self {
123                    $arg::Object(object) => Ok(object.0),
124                    $arg::Name(name) => Ok($rust::new(name)?.0),
125                }
126            }
127        }
128    };
129}

179/// A content identifier: a CIDv1 with a digest of at most 64 bytes.
180///
181/// `Cid(cid)` parses text or binary, as `from_text` or `from_bytes` does.
182/// `str` gives the text and `bytes` the binary form. CIDs compare and hash
183/// by value, and match as `Cid(codec, hash, digest)`.
184///
185/// >>> from covalence import Cid
186/// >>> cid = Cid.of(b'hello', 'sha256')
187/// >>> Cid(str(cid)) == Cid(bytes(cid)) == cid
188/// True
189/// >>> cid.codec, cid.hash
190/// (Codec('raw'), Hash('sha256'))
191#[pyclass(
192    skip_from_py_object,
193    frozen,
194    eq,
195    hash,
196    name = "Cid",
197    module = "covalence"
198)]
199#[pyo3(crate = "covalence_lib_python::pyo3")]
200#[derive(Clone, PartialEq, Eq, Hash)]
201struct PyCid {
202    inner: Cid64,
203}

276        Self::new_cid(codec.code()?, hash.code()?, &digest)

279    /// The CID of `data`, hashed with `hash` and labelled with `codec`.
280    ///
281    /// The defaults are BLAKE3 and raw. Covalence computes SHA-1, SHA-256,
282    /// SHA-512 and BLAKE3; another hash raises UnsupportedError.
283    #[staticmethod]
284    #[pyo3(signature = (data, hash = None, codec = None))]
285    fn of(
286        py: Python<'_>,
287        data: PyBackedBytes,
288        hash: Option<HashArg<'_>>,
289        codec: Option<CodecArg<'_>>,
290    ) -> PyResult<Self> {
291        let hash = hash.map_or(Ok(Multicodec::BLAKE3), |hash| hash.code())?;
292        let codec = codec.map_or(Ok(Multicodec::RAW), |codec| codec.code())?;
293        let digest = py
294            .detach(|| hash.digest(&data))
295            .ok_or_else(|| raise_as(Class::Unsupported, format!("cannot hash with {hash:?}")))?;
296        Self::new_cid(codec, hash, &digest)
297    }

304        let (_, cid, _) = py.detach(|| hash_file(&path))?;
305        Ok(Self { inner: cid.into() })
306    }

346// Nesting past `MAX_DEPTH` fails before recursing, which also stops cyclic input.
347fn feed(builder: &mut Builder, obj: &Bound<'_, PyAny>) -> PyResult<()> {
348    let mut push = |event| builder.push(event).map_err(raise);
349    if let Ok(value) = obj.extract::<PyRef<'_, PyValue>>() {
350        return value.inner.events().try_for_each(push);
351    }
352    if obj.is_none() {
353        return push(Event::Null);
354    }
355    if obj.is_instance_of::<PyBool>() {
356        return push(Event::Bool(obj.extract()?));
357    }
358    if obj.is_instance_of::<PyInt>() {
359        let value = obj
360            .extract()
361            .map_err(|_| fail(format!("integer {obj} is outside i64")))?;
362        return push(Event::Integer(value));
363    }
364    if obj.is_instance_of::<PyFloat>() {
365        return Err(raise(Error::Float));
366    }
367    if let Ok(value) = obj.cast::<PyString>() {
368        return push(Event::String(value.to_str()?));
369    }
370    if let Ok(cid) = obj.extract::<PyRef<'_, PyCid>>() {
371        return push(Event::Link(cid.dasl()?));
372    }
373    let items: Option<Vec<_>> = match obj.cast::<PyList>() {
374        Ok(list) => Some(list.iter().collect()),
375        Err(_) => obj
376            .cast::<PyTuple>()
377            .ok()
378            .map(|tuple| tuple.iter().collect()),
379    };
380    if let Some(items) = items {
381        push(Event::List(items.len()))?;
382        return items.iter().try_for_each(|item| feed(builder, item));
383    }
384    if let Ok(dict) = obj.cast::<PyDict>() {
385        let mut entries = Vec::with_capacity(dict.len());
386        for (key, value) in dict.iter() {
387            let key = key
388                .cast::<PyString>()
389                .map_err(|_| PyTypeError::new_err("map keys must be strings"))?
390                .to_str()?
391                .to_owned();
392            entries.push((key, value));
393        }
394        entries.sort_by(|(a, _), (b, _)| (a.len(), a).cmp(&(b.len(), b)));
395        push(Event::Map(entries.len()))?;
396        for (key, value) in &entries {
397            builder.push(Event::String(key)).map_err(raise)?;
398            feed(builder, value)?;
399        }
400        return Ok(());
401    }
402    let bytes = obj.extract::<PyBackedBytes>().map_err(|_| {
403        let name = obj
404            .get_type()
405            .name()
406            .map_or_else(|_| "?".into(), |name| name.to_string());
407        PyTypeError::new_err(format!("{name} cannot be a DASL value"))
408    })?;
409    push(Event::Bytes(&bytes))
410}

412fn to_native(py: Python<'_>, value: &Value) -> PyResult<Py<PyAny>> {
413    Ok(match value {
414        Value::Null => py.None(),
415        Value::Bool(value) => value.into_pyobject(py)?.to_owned().into_any().unbind(),
416        Value::Integer(value) => value.into_pyobject(py)?.into_any().unbind(),
417        Value::String(value) => value.into_pyobject(py)?.into_any().unbind(),
418        Value::Bytes(value) => PyBytes::new(py, value).into_any().unbind(),
419        Value::Link(cid) => Py::new(py, PyCid::from(*cid))?.into_any(),
420        Value::List(items) => PyList::new(
421            py,
422            items
423                .iter()
424                .map(|item| to_native(py, item))
425                .collect::<PyResult<Vec<_>>>()?,
426        )?
427        .into_any()
428        .unbind(),
429        Value::Map(map) => {
430            let dict = PyDict::new(py);
431            for (key, value) in map {
432                dict.set_item(key, to_native(py, value)?)?;
433            }
434            dict.into_any().unbind()
435        }
436    })
437}

439/// A DASL value: null, bool, integer, bytes, string, list, map or link.
440///
441/// Map keys are strings. `Value(value)` takes None, bool, int, str, bytes,
442/// `Cid`, `Value`, and lists, tuples and dicts with str keys of them. It
443/// checks the same rules as `decode`:
444///
445/// - Links must be DASL CIDs: raw or DRISL, with a SHA-256 or BLAKE3 digest.
446/// - Integers outside i64, floats and nesting deeper than 126 raise
447///   ValueError.
448///
449/// Other types raise TypeError. Values compare and hash by value.
450///
451/// >>> from covalence import Value
452/// >>> value = Value({'a': [1, b'x']})
453/// >>> Value.decode(value.encode()) == value
454/// True
455/// >>> value.to_json()
456/// '{"a":[1,{"$bytes":"eA"}]}'
457#[pyclass(
458    skip_from_py_object,
459    frozen,
460    eq,
461    hash,
462    name = "Value",
463    module = "covalence"
464)]
465#[pyo3(crate = "covalence_lib_python::pyo3")]
466#[derive(Clone, PartialEq, Eq, Hash)]
467struct PyValue {
468    inner: Value,
469}
470
471#[pymethods]
472#[pyo3(crate = "covalence_lib_python::pyo3")]
473impl PyValue {
474    #[new]
475    fn new(value: &Bound<'_, PyAny>) -> PyResult<Self> {
476        let mut builder = Builder::default();
477        feed(&mut builder, value)?;
478        Ok(Self {
479            inner: builder.finish().map_err(raise)?,
480        })
481    }
482
483    /// The kind of value, as a string.
484    ///
485    /// One of "null", "bool", "integer", "bytes", "string", "list", "map" or
486    /// "link".
487    ///
488    /// >>> from covalence import Value
489    /// >>> Value([1, 'a']).kind
490    /// 'list'
491    #[getter]
492    fn kind(&self) -> &'static str {
493        match self.inner {
494            Value::Null => "null",
495            Value::Bool(_) => "bool",
496            Value::Integer(_) => "integer",
497            Value::Bytes(_) => "bytes",
498            Value::String(_) => "string",
499            Value::List(_) => "list",
500            Value::Map(_) => "map",
501            Value::Link(_) => "link",
502        }
503    }
504
505    /// The value as None, bool, int, str, bytes, `Cid`, lists and dicts.
506    fn to_native(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
507        to_native(py, &self.inner)
508    }
509
510    /// The DRISL encoding.
511    fn encode<'py>(&self, py: Python<'py>) -> PyResult<Bound<'py, PyBytes>> {
512        let bytes = py.detach(|| self.inner.encode()).map_err(raise)?;
513        Ok(PyBytes::new(py, &bytes))
514    }
515
516    /// Decode exactly one DRISL value.
517    ///
518    /// Invalid DRISL raises ValueError.
519    #[staticmethod]
520    fn decode(py: Python<'_>, data: PyBackedBytes) -> PyResult<Self> {
521        Ok(Self {
522            inner: py.detach(|| Value::decode(&data)).map_err(raise)?,
523        })
524    }
525
526    /// The value as ATProto JSON.
527    ///
528    /// Links become `{"$link": "<CID>"}` and bytes `{"$bytes": "<base64>"}`.
529    fn to_json(&self, py: Python<'_>) -> PyResult<String> {
530        py.detach(|| self.inner.to_json()).map_err(raise)
531    }
532
533    /// Parse ATProto JSON, as `to_json` writes it.
534    ///
535    /// Invalid JSON raises ValueError.
536    ///
537    /// >>> from covalence import Value
538    /// >>> Value.from_json('{"a": {"$bytes": "eA"}}') == Value({'a': b'x'})
539    /// True
540    #[staticmethod]
541    fn from_json(py: Python<'_>, text: &str) -> PyResult<Self> {
542        Ok(Self {
543            inner: py.detach(|| Value::from_json(text)).map_err(raise)?,
544        })
545    }
546
547    /// The CID of the DRISL encoding, hashed with BLAKE3.
548    ///
549    /// This is the CID `Cas.put_value` returns.
550    fn cid(&self, py: Python<'_>) -> PyResult<PyCid> {
551        Ok(py.detach(|| self.inner.cid()).map_err(raise)?.into())
552    }
553
554    /// The CIDs the value links to, in encoding order, with repeats.
555    fn links(&self) -> Vec<PyCid> {
556        self.inner.links().map(PyCid::from).collect()
557    }
558
559    fn __repr__(&self, py: Python<'_>) -> PyResult<String> {
560        Ok(format!("Value({})", self.to_native(py)?.bind(py).repr()?))
561    }
562}

591    /// A store in S3, configured as the AWS SDKs are.
592    ///
593    /// `location` is an `s3://bucket/prefix/` URL, or a bucket name with
594    /// `prefix`. `region` and `profile` override the SDK configuration.
595    /// `concurrency` is how many 8 MiB parts of one value move at once (16 by
596    /// default; 0 counts as 1), which bounds the memory a streamed value
597    /// holds.
598    ///
599    /// - An invalid location, a URL with `prefix` too, or no region raises
600    ///   ValueError.
601    /// - Keys must be UTF-8 with non-empty `/`-separated segments other than
602    ///   `.` and `..`; other keys raise StoreError.
603    /// - Credentials are fetched at the first request; missing or failed
604    ///   credentials raise StoreError then.
605    /// - Without `s3:ListBucket`, S3 answers 403 for a missing key, so reading
606    ///   one raises StoreError instead of returning None or False.
607    #[staticmethod]
608    #[pyo3(signature = (location, prefix = None, *, region = None, profile = None, concurrency = None))]
609    fn s3(
610        py: Python<'_>,
611        location: &str,
612        prefix: Option<&str>,
613        region: Option<String>,
614        profile: Option<String>,
615        concurrency: Option<usize>,
616    ) -> PyResult<Self> {
617        let location = match (location.strip_prefix("s3://"), prefix) {
618            (None, prefix) => S3Location::new(location, prefix.unwrap_or_default()),
619            (Some(_), None) => location.parse(),
620            (Some(_), Some(_)) => {
621                return Err(fail(
622                    "pass the prefix in the s3:// URL or with a bucket name",

624            }

632            Some(parts) => kv.concurrency(parts),
633            None => kv,
634        };
635        Ok(Self {
636            inner: kv.into_dyn(),
637        })
638    }

670/// Bytes read whole or by range, not checked except by `fact`.
671///
672/// - `Blob(data)` holds a snapshot of `data` in memory.
673/// - `Blob(blob)` shares `blob`'s source.
674/// - Copies keep their class and share the source.
675///
676/// Its methods are not virtual: a Python subclass's overrides change what
677/// Python callers see, but `fact` and `slice` read the Rust blob underneath.
678///
679/// >>> from covalence import Blob
680/// >>> blob = Blob(b'hello world')
681/// >>> blob.size(), blob.slice(6, 5).read_all()
682/// (11, b'world')
683#[pyclass(subclass, frozen, name = "Blob", module = "covalence")]
684#[pyo3(crate = "covalence_lib_python::pyo3")]
685struct PyBlob {
686    inner: Arc<dyn Blob>,
687}
688
689impl PyBlob {
690    /// A blob of `blob`'s class over the same source, with its attributes
691    /// copied, or deep copied with `memo`.
692    fn copy<'py>(
693        blob: &Bound<'py, Self>,
694        memo: Option<&Bound<'py, PyAny>>,
695    ) -> PyResult<Bound<'py, PyAny>> {
696        let py = blob.py();
697        let class = blob.get_type();
698        let copy = class.call_method1(intern!(py, "__new__"), (&class, blob))?;
699        if let Ok(state) = blob.getattr(intern!(py, "__dict__")) {
700            let state = match memo {
701                Some(memo) => {
702                    memo.set_item(blob.as_ptr() as usize, &copy)?;
703                    py.import("copy")?.call_method1("deepcopy", (state, memo))?
704                }
705                None => state,
706            };
707            copy.getattr(intern!(py, "__dict__"))?
708                .call_method1("update", (state,))?;
709        }
710        Ok(copy)
711    }
712}

785    /// 'dag-cbor'

787    /// True

808/// Bytes and a CID they hash to: a `Fact`, and a blob of those bytes.
809///
810/// Facts are equal when their CIDs and bytes are, and hash by CID. This
811/// class cannot be subclassed.
812///
813/// >>> from covalence import BlobFact, Cid
814/// >>> fact = BlobFact.of(b'hello')
815/// >>> fact.cid == Cid.of(b'hello'), fact.data, fact.value()
816/// (True, b'hello', Value(b'hello'))
817#[pyclass(frozen, eq, extends = PyBlob, name = "BlobFact", module = "covalence")]

820struct PyBlobFact {
821    inner: Fact,
822}

824impl PyBlobFact {
825    /// `inner` as a Python fact, which is a blob of its bytes.
826    fn new(py: Python<'_>, inner: Fact) -> PyResult<Py<Self>> {
827        Self::over(py, Arc::new(inner.clone()), inner)
828    }
829
830    /// `inner` as a Python fact, read through `source`.
831    fn over(py: Python<'_>, source: Arc<dyn Blob>, inner: Fact) -> PyResult<Py<Self>> {
832        let blob = PyBlob { inner: source };
833        Py::new(
834            py,
835            PyClassInitializer::from(blob).add_subclass(Self { inner }),
836        )
837    }

899/// A content-addressed store, which finds bytes by their CIDs.
900///
901/// Build one with `memory`, `sqlite` or `kv`. `put` returns the raw BLAKE3
902/// CID of the bytes, and a BLAKE3 CID with another codec finds the same
903/// bytes.
904///
905/// - A CID with a digest over 32 bytes raises ValueError.
906/// - A failed check raises IntegrityError.
907/// - A failure of the storage underneath raises StoreError.
908///
909/// >>> from covalence import Cas, Cid
910/// >>> cas = Cas.memory()
911/// >>> cid = cas.put(b'hello')
912/// >>> cid == Cid.of(b'hello'), cas.get(cid)
913/// (True, b'hello')
914/// >>> cas.missing([cid, Cid.of(b'other')]) == [Cid.of(b'other')]
915/// True
916#[pyclass(frozen, name = "Cas", module = "covalence")]
917#[pyo3(crate = "covalence_lib_python::pyo3")]
918struct PyCas {
919    inner: Arc<dyn Cas>,
920}
921
922#[pymethods]

924impl PyCas {
925    /// A new, empty store in memory.
926    #[staticmethod]
927    fn memory() -> Self {
928        Self {
929            inner: MemoryCas::default().into_dyn(),
930        }
931    }

933    /// A store in the SQLite database at `path`, or in memory without one.
934    #[staticmethod]
935    #[pyo3(signature = (path = None))]
936    fn sqlite(py: Python<'_>, path: Option<PathBuf>) -> PyResult<Self> {

939                Some(path) => SqliteCas::open(path),

948    /// A store of blocks in `kv`, each under `prefix/blake3/<digest>/bytes`.
949    ///
950    /// - Without a prefix, the key is `blake3/<digest>/bytes`.
951    /// - The digest is the BLAKE3 digest in lowercase unpadded base32.
952    /// - A trailing `/` on `prefix` is optional; an invalid prefix raises
953    ///   ValueError.

957        let kv = kv.inner.clone();

959            Some(prefix) => KvCas::with_prefix(kv, prefix).map_err(fail)?,
960            None => KvCas::new(kv),
961        };
962        Ok(Self {
963            inner: store.into_dyn(),
964        })
965    }

967    /// Store `data` and return its raw BLAKE3 CID.
968    ///

970    fn put(&self, py: Python<'_>, data: PyBackedBytes) -> PyResult<PyCid> {
971        let key = self.store(py, Bytes::from_owner(data))?;
972        Ok(PyCid { inner: key.into() })
973    }

975    /// Store the DRISL encoding of `value` and return its DRISL CID.
976    ///
977    /// The CID is hashed with BLAKE3, so it is `Value(value).cid()`.
978    fn put_value(&self, py: Python<'_>, value: &Bound<'_, PyAny>) -> PyResult<PyCid> {
979        let value = PyValue::new(value)?;
980        let data = py.detach(|| value.inner.encode()).map_err(raise)?;
981        let key = self.store(py, data.into())?;

985    /// Store the file at `path` and return its raw BLAKE3 CID.
986    ///
987    /// Checks that the store keeps the file under that CID.
988    ///
989    /// - Reads the file twice, to hash it and to store it, and stores only
990    ///   the bytes it hashed. If they changed in between it raises
991    ///   IntegrityError, and if the file got shorter OSError, storing nothing.
992    /// - `Cas.kv` over `Kv.s3` streams the file; other stores read it into
993    ///   memory.
994    fn put_file(&self, py: Python<'_>, path: PathBuf) -> PyResult<PyCid> {
995        let (mut file, claim, len) = py.detach(|| hash_file(&path))?;
996        let key = py
997            .detach(|| self.inner.put_from(claim, &mut file, len))
998            .map_err(raise)?;
999        check_key(key, claim)?;
1000        Ok(PyCid { inner: key.into() })
1001    }
1002
1003    /// Store a block identified by `cid` from `chunks` and return its raw BLAKE3 CID.
1004    ///
1005    /// For bytes that arrive in pieces, such as a request body, without
1006    /// holding them all at once.
1007    ///
1008    /// - `cid` must be a BLAKE3 CID of any codec; another hash raises
1009    ///   ValueError.
1010    /// - `chunks` yields bytes or bytearray. Before a new block is stored,
1011    ///   its bytes must match `length` and `cid`, and the iterable must end there.
1012    /// - A store may return an existing block without consuming all of
1013    ///   `chunks`. Unread bytes and source errors are not checked.
1014    /// - While consuming input, a hash mismatch raises IntegrityError; too
1015    ///   few or too many bytes raises ValueError; a source exception propagates
1016    ///   unchanged. These failures publish no new block.
1017    ///   If cleanup also fails, raises StoreError with the source exception
1018    ///   as its `__cause__` and the cleanup failure in its message.
1019    /// - `Cas.kv` over `Kv.s3` streams the bytes up in parts, holding about as
1020    ///   many parts as upload at once; other stores hold them in memory.
1021    fn put_from(
1022        &self,
1023        py: Python<'_>,
1024        cid: PyRef<'_, PyCid>,
1025        chunks: &Bound<'_, PyAny>,
1026        length: u64,
1027    ) -> PyResult<PyCid> {
1028        let claim = cid
1029            .key()?
1030            .raw_blake3()
1031            .ok_or_else(|| fail(format!("{} is not a BLAKE3 CID", cid.inner)))?;
1032        let mut reader = Chunks {
1033            iter: chunks.try_iter()?.unbind(),
1034            chunk: None,
1035            offset: 0,
1036            left: length,
1037            error: None,
1038        };
1039        // No read asks past the last byte, so check that nothing follows it here.
1040        if length == 0 {
1041            reader.end()?;
1042        }
1043        let result = py.detach(|| self.inner.put_from(claim, &mut reader, length));
1044        if let Some(source) = reader.error.take() {
1045            return match result {
1046                Err(error @ covalence_cas_trait::Error::Backend { .. }) => {
1047                    let error = raise(error);
1048                    error.set_cause(py, Some(source));
1049                    Err(error)
1050                }
1051                _ => Err(source),
1052            };
1053        }
1054        let key = result.map_err(raise)?;
1055        check_key(key, claim)?;
1056        Ok(PyCid { inner: key.into() })
1057    }

1059    /// Write the bytes under `cid` to the file at `path` and return their length.
1060    ///
1061    /// Returns None if missing.
1062    ///
1063    /// - Checks the bytes against `cid` in a temporary file beside `path`,
1064    ///   which replaces `path` only once the check passes. A missing CID or a
1065    ///   failure leaves `path` as it was.
1066    /// - `Cas.kv` over `Kv.s3` streams the bytes; other stores hold them in
1067    ///   memory.
1068    fn get_file(
1069        &self,
1070        py: Python<'_>,
1071        cid: PyRef<'_, PyCid>,
1072        path: PathBuf,
1073    ) -> PyResult<Option<u64>> {
1074        let cid = cid.key()?;
1075        let (mut temp, permissions) = py.detach(|| temp_beside(&path))?;
1076        let len = py
1077            .detach(|| self.inner.get_to(&cid, temp.as_file_mut()))
1078            .map_err(raise)?;
1079        if len.is_some() {
1080            py.detach(|| {
1081                temp.as_file().set_permissions(permissions)?;
1082                temp.persist(&path).map_err(|error| error.error)
1083            })?;
1084        }
1085        Ok(len)
1086    }

1202impl PyCas {
1203    /// Store `data`, checking that the store keeps it under its BLAKE3 hash.
1204    fn store(&self, py: Python<'_>, data: Bytes) -> PyResult<Cid> {
1205        let claim = py.detach(|| *Fact::<Blake3>::of(data.clone()).claim());
1206        let key = py.detach(|| self.inner.put(data)).map_err(raise)?;
1207        check_key(key, claim)?;
1208        Ok(key)
1209    }

1212/// The bytes of a Python iterable of chunks, read as exactly `left` more
1213/// bytes. The read handing over the last of them first checks that the
1214/// iterable ends there, so a store reading it never sees a last byte that
1215/// more bytes follow. On failure it keeps the Python exception in `error`.
1216struct Chunks {
1217    iter: Py<PyIterator>,
1218    chunk: Option<PyBackedBytes>,
1219    offset: usize,
1220    left: u64,
1221    error: Option<PyErr>,
1222}
1223
1224impl Chunks {
1225    /// The next chunk, skipping empty ones, or None at the end.
1226    fn next(&self) -> PyResult<Option<PyBackedBytes>> {
1227        Python::attach(|py| {
1228            let iter = self.iter.bind(py);
1229            for item in iter.clone() {
1230                let chunk: PyBackedBytes = item?.extract()?;
1231                if !chunk.is_empty() {
1232                    return Ok(Some(chunk));
1233                }
1234            }
1235            Ok(None)
1236        })
1237    }
1238
1239    /// ValueError unless the iterable has no more bytes.
1240    fn end(&self) -> PyResult<()> {
1241        match self.next()? {
1242            None => Ok(()),
1243            Some(_) => Err(PyValueError::new_err("chunks hold more bytes than length")),
1244        }
1245    }
1246
1247    /// The next chunk, counted against `left`: ValueError if the iterable
1248    /// ends early or holds too many bytes.
1249    fn take(&mut self) -> PyResult<PyBackedBytes> {
1250        let Some(chunk) = self.next()? else {
1251            return Err(PyValueError::new_err("chunks hold fewer bytes than length"));
1252        };
1253        self.left = self
1254            .left
1255            .checked_sub(chunk.len() as u64)
1256            .ok_or_else(|| PyValueError::new_err("chunks hold more bytes than length"))?;
1257        if self.left == 0 {
1258            self.end()?;
1259        }
1260        Ok(chunk)
1261    }
1262}
1263
1264impl io::Read for Chunks {
1265    fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
1266        if buf.is_empty() {
1267            return Ok(0);
1268        }
1269        let chunk = match self.chunk.take() {
1270            Some(chunk) if self.offset < chunk.len() => chunk,
1271            _ if self.left == 0 => return Ok(0),
1272            _ => {
1273                self.offset = 0;
1274                self.take().map_err(|error| {
1275                    let message = error.to_string();
1276                    self.error = Some(error);
1277                    io::Error::other(message)
1278                })?
1279            }
1280        };
1281        let read = buf.len().min(chunk.len() - self.offset);
1282        buf[..read].copy_from_slice(&chunk[self.offset..self.offset + read]);
1283        self.offset += read;
1284        self.chunk = Some(chunk);
1285        Ok(read)
1286    }
1287}

1289/// `IntegrityError` unless the store kept the data under `claim`.
1290fn check_key(key: Cid, claim: Blake3) -> PyResult<()> {
1291    if key.same_multihash(&claim.into()) {
1292        return Ok(());
1293    }
1294    let message = format!("the store kept the data under {key}, not {claim}");
1295    Err(raise_as(Class::Integrity, message))
1296}

1298/// The file at `path`, rewound after hashing it, its raw BLAKE3 CID and its
1299/// length.
1300fn hash_file(path: &Path) -> io::Result<(File, Blake3, u64)> {
1301    let mut file = File::open(path)?;
1302    let mut hasher = blake3::Hasher::new();
1303    let len = io::copy(&mut BufReader::with_capacity(1 << 20, &file), &mut hasher)?;
1304    file.rewind()?;
1305    let digest = hasher.finalize();
1306    let cid = Blake3::new(Format::BLAKE3, digest.as_bytes()).expect("a BLAKE3 digest");
1307    Ok((file, cid, len))
1308}

1310/// A temporary file in the directory of `path` that only its owner can read,
1311/// and the permissions to give it before it replaces `path`: those of the
1312/// file at `path` if there is one, else those the umask gives a new file.
1313fn temp_beside(path: &Path) -> io::Result<(tempfile::NamedTempFile, Permissions)> {
1314    let dir = match path.parent() {
1315        Some(dir) if !dir.as_os_str().is_empty() => dir,
1316        _ => Path::new("."),
1317    };
1318    let permissions = match std::fs::metadata(path) {
1319        Ok(existing) => existing.permissions(),
1320        Err(error) if error.kind() == io::ErrorKind::NotFound => new_file_permissions(dir)?,
1321        Err(error) => return Err(error),
1322    };
1323    let temp = tempfile::Builder::new()
1324        .prefix(".covalence-")
1325        .tempfile_in(dir)?;
1326    Ok((temp, permissions))
1327}
1328
1329/// The permissions the umask gives a new file in `dir`, read from an empty
1330/// probe file that is removed at once.
1331fn new_file_permissions(dir: &Path) -> io::Result<Permissions> {
1332    let mut builder = tempfile::Builder::new();
1333    builder.prefix(".covalence-");
1334    #[cfg(unix)]
1335    builder.permissions(std::os::unix::fs::PermissionsExt::from_mode(0o666));
1336    Ok(builder
1337        .tempfile_in(dir)?
1338        .as_file()
1339        .metadata()?
1340        .permissions())
1341}
