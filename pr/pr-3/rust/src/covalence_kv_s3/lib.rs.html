1//! A key-value store on S3.
2//!
3//! [`S3Kv`] keeps the values of a key-value store as objects in an
4//! [Amazon S3](https://docs.aws.amazon.com/AmazonS3/latest/API/Welcome.html)
5//! bucket, or one on any server speaking the S3 API. An [`S3Location`] names
6//! the bucket and a key prefix within it.
7//!
8//! - [`S3Kv::new`] takes an endpoint, a region and fixed [`Credentials`]
9//!   (re-exported from `rusty-s3`).
10//! - [`S3Kv::from_aws`] finds the region, endpoint and credentials as the AWS
11//!   SDKs do.
12//! - Every request blocks, including those of the async methods.
13//!
14//! ```
15//! use covalence_kv_s3::{Credentials, Error, S3Kv, S3Location};
16//! use covalence_kv_trait::Kv;
17//! let location: S3Location = "s3://bucket/some/prefix".parse()?;
18//! assert_eq!(location.to_string(), "s3://bucket/some/prefix/");
19//! let credentials = Credentials::new("key", "secret");
20//! let store = S3Kv::new("http://localhost:9000", "us-east-1", location, credentials)?;
21//! // Keys that request URLs would rewrite fail before any request is sent.
22//! let Err(covalence_kv_trait::Error::Backend { source }) = store.get(b"a//b") else {
23//!     panic!("expected a backend error");
24//! };
25//! assert!(matches!(source.downcast_ref(), Some(Error::Key { .. })));
26//! # Ok::<(), Error>(())
27//! ```

63/// Errors from configuring and using an S3 store.
64///
65/// [`Kv`] methods return [`Error::Io`] as [`covalence_kv_trait::Error::Io`],
66/// and the rest as [`covalence_kv_trait::Error::Backend`].
67#[derive(Debug, Snafu)]
68#[snafu(crate_root(covalence_lib_error::snafu))]
69#[non_exhaustive]
70pub enum Error {
71    /// The endpoint is not a URL.
72    #[snafu(display("invalid endpoint {endpoint}: {source}"))]
73    Endpoint {
74        /// The endpoint given.
75        endpoint: String,
76        /// Why it is not a URL.
77        source: url::ParseError,
78    },
79    /// rusty-s3 rejected the bucket.
80    #[snafu(display("invalid bucket: {source}"))]
81    Bucket {
82        /// Why.
83        source: BucketError,
84    },
85    /// A request failed to complete, after any retries.
86    #[snafu(display("S3 request failed: {source}"))]
87    Http {
88        /// The last attempt's error.
89        source: ureq::Error,
90    },
91    /// A request completed with an unexpected status, after any retries.
92    #[snafu(display(
93        "S3 request failed with status {status}{}",
94        code.as_deref().map(|code| format!(" ({code})")).unwrap_or_default()
95    ))]
96    Status {
97        /// The HTTP status.
98        status: u16,
99        /// The S3 error code from the response body, if it had one.
100        code: Option<String>,
101    },
102    /// The store cannot hold `key`; see [`S3Kv`].
103    #[snafu(display(
104        "invalid key \"{}\": expected UTF-8 with non-empty '/'-separated segments other than '.' and '..'",
105        key.escape_ascii()
106    ))]
107    Key {
108        /// The key.
109        key: Vec<u8>,
110    },
111    /// The location is not `s3://bucket[/prefix/]`; see [`S3Location::new`].
112    #[snafu(display("invalid S3 location {location:?}: expected s3://bucket[/prefix/]"))]
113    Location {
114        /// The location given.
115        location: String,
116    },
117    /// A successful HEAD response lacks a valid length.
118    #[snafu(display("S3 response has no valid Content-Length"))]
119    MissingLength,
120    /// A partial response does not describe the requested bytes.
121    #[snafu(display("S3 response does not match the requested range"))]
122    RangeResponse,
123    /// A successful response lacks a field it needs, such as an ETag.
124    #[snafu(display("S3 response has no valid {field}"))]
125    MissingField {
126        /// The field.
127        field: &'static str,
128    },
129    /// The value changed between the ranged GETs reading it.
130    #[snafu(display("the S3 value changed while it was read"))]
131    Changed,
132    /// A multipart upload failed with `cause`, and so did aborting it.
133    #[snafu(display("{cause}; aborting multipart upload {upload_id} also failed: {source}"))]
134    Abort {
135        /// The upload's ID.
136        upload_id: String,
137        /// Why the upload failed.
138        cause: Box<Error>,
139        /// Why aborting it failed.
140        source: Box<Error>,
141    },
142    /// The reader of a value written, or the writer of a value read, failed.
143    #[snafu(display("{source}"))]
144    Io {
145        /// The I/O error.
146        source: io::Error,
147    },
148    /// The AWS configuration names no region.
149    #[snafu(display("no AWS region: pass one, or set AWS_REGION or a profile's region"))]
150    Region,
151    /// The AWS credential chain found no credentials, or failed to fetch them.
152    #[snafu(display("could not load AWS credentials: {source}"))]
153    Credentials {
154        /// The AWS SDK's error.
155        source: CredentialsError,
156    },
157    /// The Tokio runtime for AWS configuration could not start, or a thread
158    /// for AWS configuration and credentials or for moving parts could not
159    /// start or ended early.
160    #[snafu(display("could not run a thread: {message}"))]
161    Runtime {
162        /// What failed.
163        message: String,
164    },
165}

178/// Whether request URLs keep `key` as is: no empty, `.` or `..` segments.
179fn valid_key(key: &str) -> bool {
180    !key.split('/').any(|s| matches!(s, "" | "." | ".."))
181}

183/// A bucket and a key prefix within it, written `s3://bucket/prefix/`.
184///
185/// It parses from `s3://bucket`, `s3://bucket/` or `s3://bucket/prefix`, with
186/// or without a final `/`, and displays with one.
187#[derive(Clone, Debug, PartialEq, Eq)]
188pub struct S3Location {
189    bucket: String,
190    prefix: String,
191}
192
193impl S3Location {
194    /// The location of `prefix` in `bucket`, or [`Error::Location`]:
195    ///
196    /// - `bucket` must be non-empty, without `/`, and not `.` or `..`.
197    /// - `prefix` must be empty, or a valid key (see [`S3Kv`]) optionally
198    ///   ending in `/`.
199    pub fn new(bucket: &str, prefix: &str) -> Result<Self, Error> {
200        let location = || LocationSnafu {
201            location: format!("s3://{bucket}/{prefix}"),
202        };
203        if bucket.contains('/') || !valid_key(bucket) {
204            return location().fail();
205        }
206        let trimmed = prefix.strip_suffix('/').unwrap_or(prefix);
207        let prefix = match prefix {
208            "" => String::new(),
209            _ if valid_key(trimmed) => format!("{trimmed}/"),
210            _ => return location().fail(),
211        };
212        Ok(Self {
213            bucket: bucket.into(),
214            prefix,
215        })
216    }
217
218    /// The bucket name.
219    pub fn bucket(&self) -> &str {
220        &self.bucket
221    }
222
223    /// The key prefix: empty or ending in `/`.
224    pub fn prefix(&self) -> &str {
225        &self.prefix
226    }
227}
228
229impl FromStr for S3Location {
230    type Err = Error;
231
232    /// Parse `s3://bucket` or `s3://bucket/some/prefix/`.
233    fn from_str(s: &str) -> Result<Self, Error> {
234        let rest = s
235            .strip_prefix("s3://")
236            .context(LocationSnafu { location: s })?;
237        let (bucket, prefix) = rest.split_once('/').unwrap_or((rest, ""));
238        Self::new(bucket, prefix)
239    }
240}

242impl fmt::Display for S3Location {
243    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
244        write!(f, "s3://{}/{}", self.bucket, self.prefix)
245    }

248/// Retries after the first attempt for 5xx, 429 and transport failures.
249const RETRIES: u32 = 3;
250/// Lifetime of each signed request URL.
251const EXPIRY: Duration = Duration::from_secs(60);
252/// Limit on each request attempt, including reading the response body, on
253/// top of [`MIN_RATE`] for its body.
254const TIMEOUT: Duration = Duration::from_secs(30);
255/// The slowest transfer, in bytes per second, a request attempt waits for.
256const MIN_RATE: u64 = 1 << 20;
257/// The largest S3 object, in bytes.
258const MAX_OBJECT: u64 = 5 << 40;
259/// The most parts a multipart upload may have.
260const MAX_PARTS: u64 = 10_000;
261/// The default part size, 8 MiB: values larger are uploaded in parts of at
262/// least this size, and values are read in ranges of it.
263pub const PART: u64 = 8 << 20;
264/// The default number of parts or ranges moving at once, 16.
265pub const CONCURRENCY: usize = 16;
266/// Limit on the size of an error response body, in bytes.
267const MAX_ERROR: u64 = 64 << 10;
268/// Credentials closer than this to expiring are fetched again, leaving room
269/// for a signed URL's [`EXPIRY`] and its retries.
270const REFRESH: Duration = Duration::from_secs(300);
271
272/// Where [`S3Kv::from_aws`] looks beyond the AWS SDK's defaults.
273#[derive(Clone, Debug, Default, PartialEq, Eq)]
274pub struct AwsOptions {
275    /// The AWS profile to read the region, the endpoint and credentials from,
276    /// instead of `AWS_PROFILE` or `default`. Credentials then come only from
277    /// this profile's settings (which may name another profile or a
278    /// `credential_source`), not from the default chain.
279    pub profile: Option<String>,
280    /// The region, instead of the configured one.
281    pub region: Option<String>,
282}
283
284/// The limit on a request attempt moving `bytes` of body.
285fn timeout(bytes: u64) -> Duration {
286    TIMEOUT + Duration::from_secs(bytes.min(MAX_OBJECT) / MIN_RATE)
287}

289/// Run `future` to completion on `runtime`, on a thread of its own so that
290/// callers may already be inside an async runtime.
291fn block_on<F: Future<Output: Send> + Send>(
292    runtime: &tokio::runtime::Runtime,
293    future: F,
294) -> Result<F::Output, Error> {
295    thread::scope(|scope| {
296        thread::Builder::new()
297            .name("covalence-kv-s3-aws".into())
298            .spawn_scoped(scope, || runtime.block_on(future))
299            .map_err(|error| Error::Runtime {
300                message: error.to_string(),
301            })?
302            .join()
303            .map_err(|_| Error::Runtime {
304                message: "the thread panicked".into(),
305            })
306    })
307}

309/// A Tokio runtime that shuts down without blocking, so that a store may be
310/// dropped inside an async runtime, where dropping a runtime panics.
311struct Runtime(Option<tokio::runtime::Runtime>);
312
313impl Runtime {
314    fn new() -> Result<Self, Error> {
315        let runtime = tokio::runtime::Builder::new_current_thread()
316            .enable_all()
317            .build()
318            .map_err(|error| Error::Runtime {
319                message: error.to_string(),
320            })?;
321        Ok(Self(Some(runtime)))
322    }
323
324    fn get(&self) -> &tokio::runtime::Runtime {
325        self.0.as_ref().expect("taken only when dropped")
326    }
327}
328
329impl Drop for Runtime {
330    fn drop(&mut self) {
331        if let Some(runtime) = self.0.take() {
332            runtime.shutdown_background();
333        }
334    }
335}

337/// Where requests get their credentials.
338enum Source {
339    /// The same credentials for every request.
340    Static(Credentials),
341    /// An AWS credential provider, asked again as its credentials near expiry.
342    Provider(Box<Cached>),
343}
344
345/// An AWS credential provider and its last answer.
346struct Cached {
347    runtime: Runtime,
348    provider: SharedCredentialsProvider,
349    last: Mutex<Option<(Credentials, Option<SystemTime>)>>,
350}
351
352impl Cached {
353    fn new(runtime: Runtime, provider: SharedCredentialsProvider) -> Self {
354        Self {
355            runtime,
356            provider,
357            last: Mutex::new(None),
358        }
359    }
360
361    /// The last credentials, or new ones if there are none or they expire
362    /// within [`REFRESH`].
363    fn get(&self) -> Result<Credentials, Error> {
364        let mut last = self
365            .last
366            .lock()
367            .unwrap_or_else(|poison| poison.into_inner());
368        let fresh = |expiry: &Option<SystemTime>| {
369            expiry.is_none_or(|expiry| SystemTime::now() + REFRESH < expiry)
370        };
371        if let Some((credentials, expiry)) = &*last
372            && fresh(expiry)
373        {
374            return Ok(credentials.clone());
375        }
376        let aws = block_on(self.runtime.get(), self.provider.provide_credentials())?
377            .context(CredentialsSnafu)?;
378        let credentials = match aws.session_token() {
379            Some(token) => {
380                Credentials::new_with_token(aws.access_key_id(), aws.secret_access_key(), token)
381            }
382            None => Credentials::new(aws.access_key_id(), aws.secret_access_key()),
383        };
384        *last = Some((credentials.clone(), aws.expiry()));
385        Ok(credentials)
386    }
387}

389/// A store under an S3 location, using path-style requests
390/// (`<endpoint>/<bucket>/<key>`).
391///
392/// # Keys
393///
394/// A key is stored as the object `<prefix><key>`. Keys must be non-empty
395/// UTF-8 whose `/`-separated segments are non-empty and not `.` or `..`,
396/// since request URLs would rewrite them: `a/b` is a key, but `a//b`, `/a`,
397/// `a/` and `a/../b` are not. Other keys fail with [`Error::Key`], though
398/// other stores such as `MemoryKv` accept them.
399///
400/// # Writes
401///
402/// - A value of up to the part size ([`PART`] unless set with
403///   [`S3Kv::part_size`]) is written with one PUT.
404/// - A larger value is written with a multipart upload, in parts of the part
405///   size, or larger parts if needed to keep to S3's limit of 10,000 parts.
406/// - Each PUT or part sends the SHA-256 of its body as
407///   `x-amz-checksum-sha256`, which S3 checks against the bytes it receives.
408/// - A failed multipart upload is aborted once its parts in flight finish.
409///   If aborting fails too, the error is [`Error::Abort`].
410/// - [`Kv::put_new`] and [`Kv::put_new_from`] send `If-None-Match: *` with
411///   the PUT, or with the request completing a multipart upload, so S3 stores
412///   the value only if the key holds none (status 412 otherwise), atomically.
413///   A multipart upload that finds the key taken is aborted.
414///
415/// # Reads
416///
417/// - Values are read with ranged GETs of the part size. The first gives the
418///   value's size and ETag; a value larger than one part must have an ETag.
419/// - Every later range must report the same size and ETag, or the read fails
420///   with [`Error::Changed`], so the bytes all come from one value.
421/// - A server may ignore `Range` and send the whole value. That works for a
422///   value of up to the part size; for a larger one the read fails.
423/// - [`Kv::size`] and [`Kv::exists`] use HEAD; [`Kv::get_range`] uses one
424///   ranged GET.
425///
426/// # Requests
427///
428/// - Up to [`CONCURRENCY`] parts or ranges (unless set with
429///   [`S3Kv::concurrency`]) move at once, each on its own thread, while the
430///   reader or writer sees the bytes in order.
431/// - A request is tried up to 4 times, waiting 0.5, 1 and 2 seconds between
432///   attempts, when it gets status 429 or 5xx, or fails to complete for a
433///   reason other than a bad URL or an oversized response body.
434/// - Each attempt may take 30 seconds, plus a second per MiB of the body it
435///   sends or may receive (per part, when completing a multipart upload).
436/// - Redirects are not followed: the request fails with [`Error::Status`].
437///
438/// Every request blocks: the async methods finish theirs before returning a
439/// ready future, so do not await them on an async runtime thread.
440///
441/// ```no_run
442/// use covalence_kv_s3::{AwsOptions, S3Kv};
443/// use covalence_kv_trait::Kv;
444/// let store = S3Kv::from_aws("s3://example/packs/".parse()?, &AwsOptions::default())?;
445/// if let Some(size) = store.size(b"pack")? {
446///     let prefix = store.get_range(b"pack", 0, size.min(1024))?;
447///     // A missing key is None; an empty or exhausted range is empty bytes.
448///     if let Some(prefix) = prefix {
449///         assert!(prefix.len() <= 1024);
450///     }
451/// }
452/// # Ok::<(), Box<dyn std::error::Error>>(())
453/// ```
454pub struct S3Kv {
455    bucket: Bucket,
456    prefix: String,
457    credentials: Source,
458    agent: ureq::Agent,
459    part: u64,
460    concurrency: usize,
461}

472impl S3Kv {
473    /// A store for `location` at `endpoint`, such as
474    /// `https://s3.us-east-1.amazonaws.com`, signing requests for `region`
475    /// with `credentials`. Sends no request.
476    pub fn new(
477        endpoint: &str,
478        region: &str,
479        location: S3Location,
480        credentials: Credentials,
481    ) -> Result<Self, Error> {
482        Self::with_source(endpoint, region, location, Source::Static(credentials))
483    }
484
485    fn with_source(
486        endpoint: &str,
487        region: &str,
488        location: S3Location,
489        credentials: Source,
490    ) -> Result<Self, Error> {
491        let url = url::Url::parse(endpoint).context(EndpointSnafu { endpoint })?;
492        let bucket = Bucket::new(url, UrlStyle::Path, location.bucket, region.to_owned())
493            .context(BucketSnafu)?;
494        Ok(Self {
495            bucket,
496            prefix: location.prefix,
497            credentials,
498            agent: agent(CONCURRENCY),
499            part: PART,
500            concurrency: CONCURRENCY,
501        })
502    }
503
504    /// Write values over `bytes` in parts of at least `bytes`, and read
505    /// values in ranges of `bytes`; 0 counts as 1. S3 needs parts of at least
506    /// 5 MiB, except the last.
507    pub fn part_size(mut self, bytes: u64) -> Self {
508        self.part = bytes.max(1);
509        self
510    }
511
512    /// Move up to `parts` parts or ranges of a value at once, holding about
513    /// that many parts in memory. 1 moves them one at a time, in order.
514    pub fn concurrency(mut self, parts: usize) -> Self {
515        self.concurrency = parts.max(1);
516        self.agent = agent(self.concurrency);
517        self
518    }

520    /// A store for `location`, configured as the AWS SDKs are.
521    ///
522    /// - Profile: `options.profile`, else `AWS_PROFILE`, else `default`.
523    /// - Region: `options.region`, else `AWS_REGION`, `AWS_DEFAULT_REGION`,
524    ///   the profile, or EC2 instance metadata, in that order; else
525    ///   [`Error::Region`].
526    /// - Endpoint: `AWS_ENDPOINT_URL` or the profile's `endpoint_url`, else
527    ///   `https://s3.<region>.amazonaws.com`, or `.amazonaws.com.cn` for a
528    ///   region starting `cn-`. Other partitions need `AWS_ENDPOINT_URL`.
529    /// - Credentials: the SDKs' default chain (environment variables, profile
530    ///   files including SSO and `credential_process`, web identity, ECS and
531    ///   EKS Pod Identity, EC2 instance roles), or with `options.profile` only
532    ///   that profile's settings (which may name another profile or a
533    ///   `credential_source`).
534    ///
535    /// Credentials are fetched at the first request, and again when they are
536    /// within 5 minutes of expiring. Finding the region may query instance
537    /// metadata, which takes a moment off EC2 unless a region is set. This
538    /// may be called inside an async runtime, though it blocks.
539    pub fn from_aws(location: S3Location, options: &AwsOptions) -> Result<Self, Error> {
540        let runtime = Runtime::new()?;
541        let http = aws_smithy_http_client::Builder::new()
542            .tls_provider(Provider::Rustls(CryptoMode::Ring))
543            .build_https();
544        let mut loader = aws_config::defaults(BehaviorVersion::latest()).http_client(http.clone());
545        if let Some(region) = &options.region {
546            loader = loader.region(aws_config::Region::new(region.clone()));
547        }
548        if let Some(profile) = &options.profile {
549            let config = ProviderConfig::without_region().with_http_client(http);
550            let credentials = ProfileFileCredentialsProvider::builder()
551                .configure(&config)
552                .profile_name(profile)
553                .build();
554            loader = loader
555                .profile_name(profile)
556                .credentials_provider(credentials);
557        }
558        let config = block_on(runtime.get(), loader.load())?;
559        let region = config.region().context(RegionSnafu)?.to_string();
560        let endpoint = match config.endpoint_url() {
561            Some(endpoint) => endpoint.to_owned(),
562            // Other partitions, such as the ISO regions, need AWS_ENDPOINT_URL.
563            None if region.starts_with("cn-") => format!("https://s3.{region}.amazonaws.com.cn"),
564            None => format!("https://s3.{region}.amazonaws.com"),
565        };
566        let provider = config
567            .credentials_provider()
568            .ok_or_else(|| Error::Credentials {
569                source: CredentialsError::not_loaded("no credential provider configured"),
570            })?;
571        let source = Source::Provider(Box::new(Cached::new(runtime, provider)));
572        Self::with_source(&endpoint, &region, location, source)
573    }

575    /// The credentials to sign the next request with.
576    fn credentials(&self) -> Result<Credentials, Error> {
577        match &self.credentials {
578            Source::Static(credentials) => Ok(credentials.clone()),
579            Source::Provider(cached) => cached.get(),
580        }
581    }

583    /// The object name for `key`, which must be UTF-8 and a valid key.
584    fn object(&self, key: &[u8]) -> Result<String, Error> {
585        match str::from_utf8(key) {
586            Ok(key) if valid_key(key) => Ok(format!("{}{key}", self.prefix)),
587            _ => KeySnafu { key }.fail(),
588        }
589    }

591    /// Take `count` jobs from `jobs` in order, run `work` on each on up to
592    /// [`S3Kv::concurrency`] threads, and pass the results to `done` in job
593    /// order. A job is taken only while fewer than that many are running and
594    /// fewer than `ahead` are taken but not done, so at most that many jobs
595    /// and `ahead` results are held at once. Stops at the first failure of
596    /// `jobs`, `work` or `done`, once the work started has finished.
597    fn in_order<J: Send, R: Send>(
598        &self,
599        count: usize,
600        ahead: usize,
601        jobs: impl Iterator<Item = Result<J, Error>>,
602        work: impl Fn(J) -> Result<R, Error> + Sync,
603        mut done: impl FnMut(R) -> Result<(), Error>,
604    ) -> Result<(), Error> {
605        let window = self.concurrency;
606        let stop = AtomicBool::new(false);
607        let (job_tx, job_rx) = mpsc::channel::<(usize, J)>();
608        let job_rx = Mutex::new(job_rx);
609        thread::scope(|scope| {
610            // Dropped on the way out, even by a panic, so idle workers stop.
611            let job_tx = job_tx;
612            let (result_tx, result_rx) = mpsc::channel();
613            for _ in 0..window.min(count) {
614                let (result_tx, job_rx, work, stop) = (result_tx.clone(), &job_rx, &work, &stop);
615                let worker = move || {
616                    loop {
617                        let job = job_rx.lock().unwrap_or_else(PoisonError::into_inner).recv();
618                        let Ok((index, job)) = job else { break };
619                        if stop.load(Ordering::Relaxed) {
620                            break;
621                        }
622                        let result = panic::catch_unwind(AssertUnwindSafe(|| work(job)));
623                        if result_tx.send((index, result)).is_err() {
624                            break;
625                        }
626                    }
627                };
628                let spawned = thread::Builder::new()
629                    .name("covalence-kv-s3-part".into())
630                    .spawn_scoped(scope, worker);
631                if let Err(error) = spawned {
632                    stop.store(true, Ordering::Relaxed);
633                    return Err(Error::Runtime {
634                        message: error.to_string(),
635                    });
636                }
637            }
638            // Only workers hold senders now, so `recv` fails if all are gone.
639            drop(result_tx);
640            let mut jobs = jobs.fuse();
641            let mut pending = BTreeMap::new();
642            let (mut taken, mut returned, mut finished) = (0, 0, 0);
643            let result = (|| {
644                loop {
645                    while taken - returned < window
646                        && taken - finished < ahead
647                        && let Some(job) = jobs.next()
648                    {
649                        // The receiver lives as long as this scope.
650                        let _ = job_tx.send((taken, job?));
651                        taken += 1;
652                    }
653                    if finished == taken {
654                        return Ok(());
655                    }
656                    if let Some(result) = pending.remove(&finished) {
657                        done(result)?;
658                        finished += 1;
659                        continue;
660                    }
661                    let (index, result) = result_rx.recv().map_err(|_| Error::Runtime {
662                        message: "every part thread stopped".into(),
663                    })?;
664                    returned += 1;
665                    match result {
666                        Ok(result) => pending.insert(index, result?),
667                        Err(payload) => panic::resume_unwind(payload),
668                    };
669                }
670            })();
671            stop.store(true, Ordering::Relaxed);
672            drop(job_tx);
673            result
674        })
675    }
676
677    /// Run a request, retrying transient failures with backoff. `request`
678    /// returns the status and whatever it read from the response, so a body
679    /// that fails partway through is retried too.
680    fn send<T>(
681        &self,
682        request: impl Fn() -> Result<(u16, T), ureq::Error>,
683    ) -> Result<(u16, T), Error> {
684        let mut attempt = 0;
685        loop {
686            let result = request();
687            let transient = match &result {
688                Ok((status, _)) => *status == 429 || (500..600).contains(status),
689                Err(error) => !matches!(
690                    error,
691                    ureq::Error::BadUri(_)
692                        | ureq::Error::BodyExceedsLimit(_)
693                        | ureq::Error::InvalidProxyUrl
694                ),
695            };
696            if !transient || attempt == RETRIES {
697                return result.context(HttpSnafu);
698            }
699            thread::sleep(Duration::from_millis(500 << attempt));
700            attempt += 1;
701        }
702    }
703}

705/// An agent keeping a connection for each of `concurrency` requests.
706fn agent(concurrency: usize) -> ureq::Agent {
707    ureq::Agent::config_builder()
708        .http_status_as_error(false)
709        .max_redirects(0)
710        .timeout_global(Some(TIMEOUT))
711        .max_idle_connections(concurrency)
712        .max_idle_connections_per_host(concurrency)
713        .build()
714        .into()
715}

717/// The status and body of a response, read into `buffer`, reading at most
718/// `limit` bytes of a 200 or 206 body and [`MAX_ERROR`] bytes of any other.
719fn body(
720    mut response: ureq::http::Response<ureq::Body>,
721    limit: u64,
722    mut body: Vec<u8>,
723) -> Result<(u16, Vec<u8>), ureq::Error> {
724    let status = response.status().as_u16();
725    let limit = if matches!(status, 200 | 206) {
726        limit
727    } else {
728        MAX_ERROR
729    };
730    // Reading into a buffer of the announced size, up to `limit`, keeps a
731    // part from taking up to twice its size as the buffer grows.
732    let announced = response
733        .headers()
734        .get("content-length")
735        .and_then(|len| len.to_str().ok()?.parse::<u64>().ok())
736        .unwrap_or(0)
737        .min(limit);
738    body.clear();
739    body.reserve(usize::try_from(announced).unwrap_or(0));
740    // ureq fails once it has read `limit` bytes, even at the end of the body.
741    let config = response.body_mut().with_config();
742    let mut reader = config.limit(limit.saturating_add(1)).reader();
743    reader.read_to_end(&mut body)?;
744    Ok((status, body))
745}

747/// [`Error::Status`], with the `<Code>` of the S3 error `body` if it has one.
748fn status_error<T>(status: u16, body: &[u8]) -> Result<T, Error> {
749    let code = str::from_utf8(body)
750        .ok()
751        .and_then(|body| body.split_once("<Code>"))
752        .and_then(|(_, rest)| rest.split_once("</Code>"))
753        .map(|(code, _)| code.to_owned());
754    StatusSnafu { status, code }.fail()
755}

757impl Kv for S3Kv {
758    fn get(&self, key: &[u8]) -> Result<Option<Bytes>, covalence_kv_trait::Error> {
759        Ok(self.read(key)?)
760    }
761
762    /// Query the object size with HEAD.
763    ///
764    /// - Returns `None` on 404, and zero for an empty object.
765    /// - Requires a valid Content-Length on a successful response, else
766    ///   [`Error::MissingLength`].
767    /// - Does not download the value.
768    fn size(&self, key: &[u8]) -> Result<Option<u64>, covalence_kv_trait::Error> {
769        Ok(self.read_size(key)?)
770    }
771
772    /// Fetch a single range using a signed Range header.
773    ///
774    /// - Follows [`Kv::get_range`], including its missing/empty/EOF rules.
775    /// - A zero-length range asks HEAD, to tell a missing key from an empty range.
776    /// - Validates Content-Range and the partial body length.
777    /// - Limits the reply to `len`.
778    /// - A whole-value 200 reply is sliced, if it is within that limit.
779    /// - Servers reject offsets past `i64::MAX`: a start past it asks HEAD, as
780    ///   an empty range does, and an end past it is clipped to it.
781    /// - Rejects larger or differently positioned replies; no full-fetch fallback.
782    fn get_range(
783        &self,
784        key: &[u8],
785        start: u64,
786        len: u64,
787    ) -> Result<Option<Bytes>, covalence_kv_trait::Error> {
788        Ok(self.read_range(key, start, len)?)
789    }
790
791    fn put(&self, key: &[u8], value: Bytes) -> Result<(), covalence_kv_trait::Error> {
792        self.write(key, value, Write::Replace)?;
793        Ok(())
794    }
795
796    /// Stream the value up in parts, reading them in order and holding about
797    /// as many parts in memory as move at once.
798    fn put_from(
799        &self,
800        key: &[u8],
801        reader: &mut dyn io::Read,
802        len: u64,
803    ) -> Result<(), covalence_kv_trait::Error> {
804        self.write_from(key, reader, len, Write::Replace)?;
805        Ok(())
806    }
807
808    /// A conditional write: `If-None-Match: *`, atomic in S3. Needs only
809    /// `s3:PutObject`.
810    fn put_new(&self, key: &[u8], value: Bytes) -> Result<bool, covalence_kv_trait::Error> {
811        Ok(self.write(key, value, Write::New)?)
812    }
813
814    /// As [`Kv::put_from`], with a conditional write: `If-None-Match: *` on
815    /// the PUT or on completing the multipart upload, atomic in S3. A large
816    /// value is uploaded in full before S3 decides; when the key is taken,
817    /// the upload is aborted (`s3:AbortMultipartUpload`).
818    fn put_new_from(
819        &self,
820        key: &[u8],
821        reader: &mut dyn io::Read,
822        len: u64,
823    ) -> Result<bool, covalence_kv_trait::Error> {
824        Ok(self.write_from(key, reader, len, Write::New)?)
825    }
826
827    /// Stream the value down in ranges, writing them in order and holding
828    /// about as many ranges in memory as move at once.
829    fn get_to(
830        &self,
831        key: &[u8],
832        writer: &mut dyn io::Write,
833    ) -> Result<Option<u64>, covalence_kv_trait::Error> {
834        Ok(self.read_to(key, writer)?)
835    }
836
837    fn aget<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, Option<Bytes>> {
838        ready(self.get(key))
839    }
840
841    /// Query the size before returning a ready future.
842    ///
843    /// This blocks just like [`Kv::size`]; keep it off async runtime threads.
844    fn asize<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, Option<u64>> {
845        ready(self.size(key))
846    }
847
848    /// Fetch the range before returning a ready future.
849    ///
850    /// This blocks just like [`Kv::get_range`]; keep it off async runtime threads.
851    fn aget_range<'a>(
852        &'a self,
853        key: &'a [u8],
854        start: u64,
855        len: u64,
856    ) -> KvFuture<'a, Option<Bytes>> {
857        ready(self.get_range(key, start, len))
858    }
859
860    fn aput<'a>(&'a self, key: &'a [u8], value: Bytes) -> KvFuture<'a, ()> {
861        ready(self.put(key, value))
862    }
863
864    /// Write before returning a ready future, as [`Kv::put_new`] does.
865    ///
866    /// This blocks just like [`Kv::put_new`]; keep it off async runtime threads.
867    fn aput_new<'a>(&'a self, key: &'a [u8], value: Bytes) -> KvFuture<'a, bool> {
868        ready(self.put_new(key, value))
869    }
870
871    /// Ask with HEAD, as [`Kv::size`] does.
872    fn exists(&self, key: &[u8]) -> Result<bool, covalence_kv_trait::Error> {
873        Ok(self.read_size(key)?.is_some())
874    }
875
876    fn aexists<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, bool> {
877        ready(self.exists(key))
878    }

883    use std::{
884        sync::{
885            Arc,
886            atomic::{AtomicUsize, Ordering},
887        },
888        time::{Duration, SystemTime},
889    };
890
891    use aws_credential_types::provider::{
892        ProvideCredentials, SharedCredentialsProvider, future::ProvideCredentials as Future,
893    };
894
895    use super::{Cached, REFRESH, Runtime};
896
897    /// A provider counting its calls, whose credentials last `lifetime`.
898    #[derive(Debug)]
899    struct Counting {
900        calls: Arc<AtomicUsize>,
901        lifetime: Option<Duration>,
902    }
903
904    impl ProvideCredentials for Counting {
905        fn provide_credentials<'a>(&'a self) -> Future<'a>
906        where
907            Self: 'a,
908        {
909            let call = self.calls.fetch_add(1, Ordering::SeqCst);
910            let expiry = self.lifetime.map(|lifetime| SystemTime::now() + lifetime);
911            Future::ready(Ok(aws_credential_types::Credentials::new(
912                format!("key{call}"),
913                "secret",
914                Some("token".into()),
915                expiry,
916                "test",
917            )))
918        }
919    }
920
921    /// Credentials from a [`Counting`] provider, and how often it was asked.
922    fn fetch(lifetime: Option<Duration>, times: usize) -> (Vec<String>, usize) {
923        let calls = Arc::new(AtomicUsize::new(0));
924        let provider = Counting {
925            calls: calls.clone(),
926            lifetime,
927        };
928        let cached = Cached::new(
929            Runtime::new().unwrap(),
930            SharedCredentialsProvider::new(provider),
931        );
932        let keys = (0..times)
933            .map(|_| {
934                let credentials = cached.get().unwrap();
935                assert_eq!(credentials.token(), Some("token"));
936                credentials.key().to_owned()
937            })
938            .collect();
939        (keys, calls.load(Ordering::SeqCst))
940    }
941
942    #[test]
943    fn credentials_are_kept_until_near_expiry() {
944        assert_eq!(fetch(None, 3), (vec!["key0".into(); 3], 1));
945        let lasting = Some(REFRESH * 2);
946        assert_eq!(fetch(lasting, 3), (vec!["key0".into(); 3], 1));
947        let expiring = Some(REFRESH / 2);
948        let keys = vec!["key0".into(), "key1".into(), "key2".into()];
949        assert_eq!(fetch(expiring, 3), (keys, 3));
950    }

952    #[test]
953    fn credentials_load_inside_an_async_runtime() {
954        let runtime = tokio::runtime::Builder::new_current_thread()
955            .build()
956            .unwrap();
957        let (keys, calls) = runtime.block_on(async { fetch(None, 1) });
958        assert_eq!((keys, calls), (vec!["key0".into()], 1));
959    }
