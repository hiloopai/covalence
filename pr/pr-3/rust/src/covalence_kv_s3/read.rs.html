14/// The reply to a ranged GET.
15enum Reply {
16    /// The requested bytes, the size of the value and its ETag.
17    Partial {
18        body: Bytes,
19        size: u64,
20        etag: Option<String>,
21    },
22    /// The server ignored the range and sent the whole value.
23    Whole(Bytes),
24    /// The range starts at or past the end of the value.
25    Unsatisfiable,
26    /// There is no value.
27    Missing,
28}

30impl S3Kv {
31    /// The value under `key`, or `None` if missing.
32    pub(crate) fn read(&self, key: &[u8]) -> Result<Option<Bytes>, Error> {
33        let mut value = Vec::new();
34        let size = self.read_parts(key, |part| {
35            value.extend_from_slice(part);
36            Ok(())
37        })?;
38        Ok(size.map(|_| value.into()))
39    }
40
41    /// Write the value under `key` to `writer`, returning its size, or
42    /// `None` if missing.
43    pub(crate) fn read_to(&self, key: &[u8], writer: &mut dyn Write) -> Result<Option<u64>, Error> {
44        self.read_parts(key, |part| writer.write_all(part).context(IoSnafu))
45    }

47    /// Read the value under `key` with ranged GETs of the part size, passing
48    /// each part to `sink` in order, and return the value's size, or `None`
49    /// if missing. The first range gives the size; the rest are fetched
50    /// concurrently. Every part must come from the same value: one with the
51    /// same size and ETag, which a value of more than one part must have.
52    fn read_parts(
53        &self,
54        key: &[u8],
55        mut sink: impl FnMut(&[u8]) -> Result<(), Error>,
56    ) -> Result<Option<u64>, Error> {
57        let object = self.object(key)?;
58        let part = self.part;
59        let (size, etag) = match self.fetch(&object, (0, part - 1), part, Vec::new())? {
60            Reply::Missing => return Ok(None),
61            Reply::Unsatisfiable => return Ok(Some(0)),
62            Reply::Whole(body) => {
63                sink(&body)?;
64                return Ok(Some(body.len() as u64));
65            }
66            Reply::Partial { body, size, .. } if body.len() as u64 == size => {
67                sink(&body)?;
68                return Ok(Some(size));
69            }
70            Reply::Partial { body, size, etag } => {
71                // Without an ETag, two values of one size look alike.
72                let etag = etag.context(MissingFieldSnafu { field: "ETag" })?;
73                sink(&body)?;
74                (size, etag)
75            }
76        };
77        // The first range was a whole part, so there are fewer ranges left
78        // than bytes in the value.
79        let count = (size - part).div_ceil(part);
80        // Buffers come from this thread, which frees them once written: had
81        // the part threads allocate them, glibc would keep up to twice the
82        // ranges in flight in their arenas.
83        let ranges = (1..=count).map(|index| {
84            let start = index * part;
85            let len = part.min(size - start);
86            Ok((start, Vec::with_capacity(usize::try_from(len).unwrap_or(0))))
87        });
88        let range = |(start, buffer): (u64, Vec<u8>)| {
89            let end = start.saturating_add(part - 1);
90            match self.fetch(&object, (start, end), part, buffer)? {
91                Reply::Partial {
92                    body,
93                    size: other,
94                    etag: Some(other_etag),
95                } if other == size && other_etag == etag => Ok(body),
96                _ => ChangedSnafu.fail(),
97            }
98        };
99        let count = usize::try_from(count).unwrap_or(usize::MAX);
100        // Ranges wait their turn in memory, so take no more than run at once.
101        self.in_order(count, self.concurrency, ranges, range, |body| sink(&body))?;
102        Ok(Some(size))
103    }

105    pub(crate) fn read_size(&self, key: &[u8]) -> Result<Option<u64>, Error> {
106        let object = self.object(key)?;
107        let credentials = self.credentials()?;
108        let (status, len) = self.send(|| {
109            let url = self
110                .bucket
111                .head_object(Some(&credentials), &object)
112                .sign(EXPIRY);
113            let response = self.agent.head(url.as_str()).call()?;
114            let len = response
115                .headers()
116                .get("content-length")
117                .and_then(|len| len.to_str().ok()?.parse::<u64>().ok());
118            Ok((response.status().as_u16(), len))
119        })?;
120        match status {
121            200 => Ok(Some(len.context(MissingLengthSnafu)?)),
122            404 => Ok(None),
123            status => StatusSnafu { status, code: None }.fail(),
124        }
125    }

127    pub(crate) fn read_range(
128        &self,
129        key: &[u8],
130        start: u64,
131        len: u64,
132    ) -> Result<Option<Bytes>, Error> {
133        let object = self.object(key)?;
134        // S3 sizes fit in i64, and servers reject larger offsets. An empty
135        // range, or one past any size, cannot be asked for, so ask whether
136        // the key exists.
137        const MAX: u64 = i64::MAX as u64;
138        if len == 0 || start > MAX {
139            return Ok(self.read_size(key)?.map(|_| Bytes::new()));
140        }
141        let end = start.saturating_add(len - 1).min(MAX);
142        Ok(match self.fetch(&object, (start, end), len, Vec::new())? {
143            Reply::Partial { body, .. } => Some(body),
144            Reply::Whole(body) => {
145                let first = start.min(body.len() as u64) as usize;
146                Some(body.slice(first..))
147            }
148            Reply::Unsatisfiable => Some(Bytes::new()),
149            Reply::Missing => None,
150        })
151    }

153    /// GET bytes `start..=end` of `object`, reading at most `limit` bytes
154    /// into `buffer`.
155    fn fetch(
156        &self,
157        object: &str,
158        (start, end): (u64, u64),
159        limit: u64,
160        buffer: Vec<u8>,
161    ) -> Result<Reply, Error> {
162        // A retry reads into a new buffer, as the old one went with the failure.
163        let buffer = Cell::new(buffer);
164        let range = format!("bytes={start}-{end}");
165        let credentials = self.credentials()?;
166        let (status, (body, content_range, etag)) = self.send(|| {
167            let mut action = self.bucket.get_object(Some(&credentials), object);
168            action.headers_mut().insert("range", &range);
169            let url = action.sign(EXPIRY);
170            let request = self.agent.get(url.as_str()).config();
171            let request = request.timeout_global(Some(timeout(limit))).build();
172            let response = request.header("Range", &range).call()?;
173            let header = |name| {
174                response
175                    .headers()
176                    .get(name)
177                    .and_then(|value| value.to_str().ok())
178                    .map(str::to_owned)
179            };
180            let (content_range, etag) = (header("content-range"), header("etag"));
181            let (status, body) = body(response, limit, buffer.take())?;
182            Ok((status, (body, content_range, etag)))
183        })?;
184        match status {
185            206 => {
186                let len = body.len() as u64;
187                let size = matches_range(content_range.as_deref(), start, end, len)
188                    .context(RangeResponseSnafu)?;
189                Ok(Reply::Partial {
190                    body: body.into(),
191                    size,
192                    etag,
193                })
194            }
195            // A server may ignore Range and return the whole value, which
196            // the read limit bounds.
197            200 => Ok(Reply::Whole(body.into())),
198            416 => Ok(Reply::Unsatisfiable),
199            404 => Ok(Reply::Missing),
200            status => status_error(status, &body),
201        }
202    }
203}

205/// The size of the value, if `header` describes a single S3 range that
206/// starts at the requested offset and ends at either the requested end or
207/// EOF, and the body holds precisely those `len` bytes.
208fn matches_range(header: Option<&str>, start: u64, end: u64, len: u64) -> Option<u64> {
209    let (range, size) = header?.strip_prefix("bytes ")?.split_once('/')?;
210    let (first, last) = range.split_once('-')?;
211    let first = first.parse::<u64>().ok()?;
212    let last = last.parse::<u64>().ok()?;
213    let size = size.parse::<u64>().ok()?;
214    (first == start
215        && first <= last
216        && last < size
217        && last == end.min(size - 1)
218        && len == last - first + 1)
219        .then_some(size)
220}
