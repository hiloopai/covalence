10const CID_TAG: u64 = 42;
11
12pub(crate) fn write<W: Write>(out: &mut W, event: Event<'_>) -> io::Result<()> {
13    match event {
14        Event::Null => out.write_all(&[0xf6]),
15        Event::Bool(value) => out.write_all(&[0xf4 | u8::from(value)]),
16        Event::Integer(value @ 0..) => head(out, 0, value as u64),
17        Event::Integer(value) => head(out, 1, !value as u64),
18        Event::Bytes(value) => {
19            head(out, 2, value.len() as u64)?;
20            out.write_all(value)
21        }
22        Event::String(value) => {
23            head(out, 3, value.len() as u64)?;
24            out.write_all(value.as_bytes())
25        }
26        Event::List(len) => head(out, 4, len as u64),
27        Event::Map(len) => head(out, 5, len as u64),
28        Event::Link(cid) => {
29            let cid = cid.to_bytes();
30            head(out, 6, CID_TAG)?;
31            head(out, 2, cid.len() as u64 + 1)?;
32            out.write_all(&[0])?;
33            out.write_all(&cid)
34        }
35    }
36}
37
38fn head<W: Write>(out: &mut W, major: u8, argument: u64) -> io::Result<()> {
39    let major = major << 5;
40    match argument {
41        ..24 => out.write_all(&[major | argument as u8]),
42        24..=0xff => out.write_all(&[major | 24, argument as u8]),
43        0x100..=0xffff => {
44            out.write_all(&[major | 25])?;
45            out.write_all(&(argument as u16).to_be_bytes())
46        }
47        0x1_0000..=0xffff_ffff => {
48            out.write_all(&[major | 26])?;
49            out.write_all(&(argument as u32).to_be_bytes())
50        }
51        _ => {
52            out.write_all(&[major | 27])?;
53            out.write_all(&argument.to_be_bytes())
54        }
55    }
56}

58/// Reads minimal DRISL heads as events. Map key order is checked by the builder.
59pub(crate) struct Reader<'a>(&'a [u8]);
60
61impl<'a> Reader<'a> {
62    pub(crate) fn new(bytes: &'a [u8]) -> Self {
63        Reader(bytes)
64    }
65
66    fn event(&mut self) -> Result<Event<'a>, Error> {
67        let [byte] = self.array()?;
68        let (major, info) = (byte >> 5, byte & 0x1f);
69        if major == 7 {
70            return match info {
71                20 => Ok(Event::Bool(false)),
72                21 => Ok(Event::Bool(true)),
73                22 => Ok(Event::Null),
74                25..=27 => FloatSnafu.fail(),
75                _ => malformed("unsupported simple value"),
76            };
77        }
78        let argument = self.argument(info)?;
79        Ok(match major {
80            0 => Event::Integer(integer(argument)?),
81            1 => Event::Integer(!integer(argument)?),
82            2 => Event::Bytes(self.take(argument)?),
83            3 => Event::String(str::from_utf8(self.take(argument)?).ok().context(
84                MalformedSnafu {
85                    message: "string is not UTF-8",
86                },
87            )?),
88            4 => Event::List(length(argument)?),
89            5 => Event::Map(length(argument)?),
90            _ => Event::Link(self.link(argument)?),
91        })
92    }
93
94    /// The CID after tag `tag`: a byte string holding a zero byte and a binary CID.
95    fn link(&mut self, tag: u64) -> Result<Cid<Bdasl>, Error> {
96        ensure!(
97            tag == CID_TAG,
98            MalformedSnafu {
99                message: "unsupported tag"
100            }
101        );
102        let [byte] = self.array()?;
103        ensure!(
104            byte >> 5 == 2,
105            MalformedSnafu {
106                message: "a CID must be a byte string"
107            }
108        );
109        let argument = self.argument(byte & 0x1f)?;
110        let [0, cid @ ..] = self.take(argument)? else {
111            return malformed("a CID must start with a zero byte");
112        };
113        Ok(Cid::from_bytes(cid)?)
114    }
115
116    fn argument(&mut self, info: u8) -> Result<u64, Error> {
117        let (argument, least) = match info {
118            ..24 => return Ok(info.into()),
119            24 => (u8::from_be_bytes(self.array()?).into(), 24),
120            25 => (u16::from_be_bytes(self.array()?).into(), 0x100),
121            26 => (u32::from_be_bytes(self.array()?).into(), 0x1_0000),
122            27 => (u64::from_be_bytes(self.array()?), 0x1_0000_0000),
123            _ => return malformed("indefinite length or reserved head"),
124        };
125        ensure!(
126            argument >= least,
127            MalformedSnafu {
128                message: "head is not minimal"
129            }
130        );
131        Ok(argument)
132    }
133
134    fn array<const N: usize>(&mut self) -> Result<[u8; N], Error> {
135        Ok(self.take(N as u64)?.try_into().expect("took N bytes"))
136    }
137
138    fn take(&mut self, len: u64) -> Result<&'a [u8], Error> {
139        let (taken, rest) = usize::try_from(len)
140            .ok()
141            .and_then(|len| self.0.split_at_checked(len))
142            .context(MalformedSnafu {
143                message: "unexpected end of input",
144            })?;
145        self.0 = rest;
146        Ok(taken)
147    }
148}
149
150impl<'a> Iterator for Reader<'a> {
151    type Item = Result<Event<'a>, Error>;
152
153    fn next(&mut self) -> Option<Self::Item> {
154        (!self.0.is_empty()).then(|| self.event())
155    }
156}
157
158fn integer(argument: u64) -> Result<i64, Error> {
159    i64::try_from(argument).ok().context(MalformedSnafu {
160        message: "integer is outside i64",
161    })
162}
163
164fn length(argument: u64) -> Result<usize, Error> {
165    usize::try_from(argument).ok().context(MalformedSnafu {
166        message: "length does not fit in memory",
167    })
168}
169
170fn malformed<T>(message: &'static str) -> Result<T, Error> {
171    MalformedSnafu { message }.fail()
172}
