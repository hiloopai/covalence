1//! A content-addressed store over a key-value store.
2//!
3//! [`KvCas`] keeps blocks in any [`Kv`], such as `covalence-kv-memory`'s
4//! `MemoryKv` or `covalence-kv-s3`'s `S3Kv`. Each block lies under
5//! `blake3/<digest>/bytes`, after an optional prefix, where `<digest>` is its
6//! [BLAKE3](https://github.com/BLAKE3-team/BLAKE3-specs) digest in lowercase,
7//! unpadded [base32](https://www.rfc-editor.org/rfc/rfc4648#section-6).
8//!
9//! ```
10//! use covalence_cas_kv::KvCas;
11//! use covalence_cas_trait::{GetExt, Put};
12//! use covalence_kv_memory::MemoryKv;
13//! use covalence_kv_trait::Kv;
14//! use covalence_lib_cid::Cid;
15//! let kv = MemoryKv::default().into_dyn();
16//! let store = KvCas::with_prefix(kv.clone(), "cas")?;
17//! let cid: Cid = store.put("hello".into())?;
18//! assert_eq!(store.get_fact(&cid)?.unwrap().bytes(), "hello");
19//! // BLAKE3("hello") is ea8f163d...4a67200f in hex.
20//! let key = "cas/blake3/5khrmpntq2bjexseshc6ldklwnig56gbj23yvbxjbdcwestheahq/bytes";
21//! assert_eq!(kv.get(key.as_bytes())?.unwrap(), "hello");
22//! # Ok::<(), Box<dyn std::error::Error>>(())
23//! ```
24#![deny(missing_docs)]

41/// A store of blocks in a [`Kv`]; see the [crate] docs for the key layout.
42///
43/// - A CID with a 32-byte BLAKE3 digest finds its bytes whatever its codec;
44///   other CIDs find nothing. [`Put`] returns the raw BLAKE3 CID.
45/// - The key-value store may be remote or shared, so `get`, `aget`,
46///   `get_fact` and `aget_fact` check the bytes against the CID and fail with
47///   [`Error::Check`] if they do not match. `get_unchecked`, sizes and ranges
48///   are not checked.
49/// - `has` asks the key-value store whether the key exists, without reading
50///   the bytes.
51/// - `put`, `aput` and `put_from` never replace a block: they store through
52///   the key-value store's [`Kv::put_new`], [`Kv::aput_new`] and
53///   [`Kv::put_new_from`], so concurrent writers of one block race safely.
54///   Writes fail if the backend does not implement atomic create-only writes.
55/// - `put_from` and `get_to` stream through the key-value store's own
56///   `put_new_from` and `get_to`, hashing the bytes as they pass.
57///   If the backend skips an existing block, `put_from` returns without
58///   consuming or checking any remaining input.
59/// - If the bytes `put_from` reads do not hash to the CID, the read that would
60///   hand over the last byte fails instead. The key-value store's `put_from`
61///   then fails as [`Kv::put_from`] says, storing nothing, and `put_from`
62///   fails with [`Error::Check`]; or with [`Error::Backend`] if the key-value
63///   store reports a further failure, such as failing to clean up.
64/// - [`Cas::info`] is `{"name": "kv", "prefix": <prefix>, "kv": <the
65///   key-value store's info>}`; [`Cas::stats`] is the key-value store's.
66///
67/// # S3
68///
69/// Writes use conditional requests to preserve existing blocks. They do not
70/// make a preliminary HEAD request, so they need only write access.
71///
72/// `has` uses a HEAD request. S3 answers a HEAD for a missing key with 403 to
73/// a client that may not list the bucket.
74///
75/// S3 may consume the source before discovering that a block already exists.
76/// Once the backend skips insertion, `put_from` leaves any remaining input
77/// unread.
78pub struct KvCas {
79    kv: Arc<dyn Kv>,
80    prefix: String,
81}

83/// A prefix that is not `/`-separated, non-empty segments other than `.` and
84/// `..`, optionally ending in `/`.
85#[derive(Debug, Snafu)]
86#[snafu(crate_root(covalence_lib_error::snafu))]
87#[snafu(display("invalid key prefix {prefix:?}"))]
88pub struct InvalidPrefix {
89    prefix: String,
90}

92impl KvCas {
93    /// A store with keys at the root of `kv`.
94    pub fn new(kv: Arc<dyn Kv>) -> Self {
95        Self {
96            kv,
97            prefix: String::new(),
98        }
99    }
100
101    /// A store with keys under `prefix/` in `kv`; a trailing `/` on `prefix`
102    /// is optional.
103    ///
104    /// Fails with [`InvalidPrefix`] unless `prefix` is `/`-separated,
105    /// non-empty segments other than `.` and `..`, so the empty prefix fails:
106    /// use [`KvCas::new`] instead.
107    ///
108    /// ```
109    /// use covalence_cas_kv::KvCas;
110    /// use covalence_kv_memory::MemoryKv;
111    /// use covalence_kv_trait::Kv;
112    /// let kv = MemoryKv::default().into_dyn();
113    /// assert!(KvCas::with_prefix(kv.clone(), "a/b/").is_ok());
114    /// for prefix in ["", "/", "/a", "a//b", "a/../b"] {
115    ///     assert!(KvCas::with_prefix(kv.clone(), prefix).is_err());
116    /// }
117    /// ```
118    pub fn with_prefix(kv: Arc<dyn Kv>, prefix: &str) -> Result<Self, InvalidPrefix> {
119        let trimmed = prefix.strip_suffix('/').unwrap_or(prefix);
120        if trimmed.split('/').any(|s| matches!(s, "" | "." | "..")) {
121            return InvalidPrefixSnafu { prefix }.fail();
122        }
123        Ok(Self {
124            kv,
125            prefix: format!("{trimmed}/"),
126        })
127    }
128
129    /// The key for `cid`.
130    fn key(&self, cid: &Blake3) -> String {
131        let digest = BASE32_NOPAD.encode(cid.digest()).to_ascii_lowercase();
132        format!("{}blake3/{digest}/bytes", self.prefix)
133    }
134}

136fn backend(error: covalence_kv_trait::Error) -> Error {
137    match error {
138        covalence_kv_trait::Error::Io { source } => Error::Io { source },
139        error => Error::Backend {
140            source: Box::new(error),
141        },
142    }
143}

145/// Whether `hasher` has hashed exactly the bytes of `cid`.
146fn hashes_to(hasher: &blake3::Hasher, cid: &Blake3) -> bool {
147    hasher.finalize().as_bytes() == cid.digest()
148}

150/// The next `left` bytes of `reader`, which must hash to `cid`: the read
151/// returning the last of them fails instead if they do not.
152struct Checked<'a> {
153    reader: &'a mut dyn Read,
154    hasher: blake3::Hasher,
155    left: u64,
156    cid: Blake3,
157    mismatch: bool,
158}
159
160impl Read for Checked<'_> {
161    fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
162        let max = buf
163            .len()
164            .min(usize::try_from(self.left).unwrap_or(usize::MAX));
165        if max == 0 {
166            return Ok(0);
167        }
168        let read = self.reader.read(&mut buf[..max])?;
169        self.hasher.update(&buf[..read]);
170        self.left -= read as u64;
171        if self.left == 0 && !hashes_to(&self.hasher, &self.cid) {
172            self.mismatch = true;
173            let error = covalence_lib_cid::Error::Mismatch;
174            return Err(io::Error::new(io::ErrorKind::InvalidData, error));
175        }
176        Ok(read)
177    }
178}

180/// A writer hashing what it passes on.
181struct Hashing<'a> {
182    writer: &'a mut dyn Write,
183    hasher: blake3::Hasher,
184}
185
186impl Write for Hashing<'_> {
187    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
188        let written = self.writer.write(buf)?;
189        self.hasher.update(&buf[..written]);
190        Ok(written)
191    }
192
193    fn flush(&mut self) -> io::Result<()> {
194        self.writer.flush()
195    }
196}

321impl Cas for KvCas {
322    fn has(&self, key: &Cid) -> Result<bool, Error> {
323        key.raw_blake3().map_or(Ok(false), |key| {
324            let key = self.key(&key);
325            self.kv.exists(key.as_bytes()).map_err(backend)
326        })
327    }
328
329    /// The name `kv`, the key prefix, and the key-value store's info.
330    fn info(&self) -> Result<Value, Error> {
331        Ok(map([
332            ("name", Value::String("kv".into())),
333            ("prefix", Value::String(self.prefix.as_str().into())),
334            ("kv", self.kv.info().map_err(backend)?),
335        ]))
336    }
337
338    fn stats(&self) -> Result<Value, Error> {
339        self.kv.stats().map_err(backend)
340    }
341
342    fn put_from(&self, cid: Blake3, reader: &mut dyn Read, len: u64) -> Result<Cid, Error> {
343        // No read hands over the last byte of nothing, so check it here.
344        if len == 0 && !hashes_to(&blake3::Hasher::new(), &cid) {
345            return Err(covalence_lib_cid::Error::Mismatch.into());
346        }
347        let mut checked = Checked {
348            reader,
349            hasher: blake3::Hasher::new(),
350            left: len,
351            cid,
352            mismatch: false,
353        };
354        let key = self.key(&cid);
355        match self.kv.put_new_from(key.as_bytes(), &mut checked, len) {
356            Ok(_) => Ok(cid.into()),
357            // The store failed as asked; a store error beyond that, such as a
358            // failed cleanup, is the one to report.
359            Err(covalence_kv_trait::Error::Io { .. }) if checked.mismatch => {
360                Err(covalence_lib_cid::Error::Mismatch.into())
361            }
362            Err(error) => Err(backend(error)),
363        }
364    }
365
366    fn get_to(&self, key: &Cid, writer: &mut dyn Write) -> Result<Option<u64>, Error> {
367        let Some(cid) = key.raw_blake3() else {
368            return Ok(None);
369        };
370        let mut hashing = Hashing {
371            writer,
372            hasher: blake3::Hasher::new(),
373        };
374        let key = self.key(&cid);
375        let Some(len) = self
376            .kv
377            .get_to(key.as_bytes(), &mut hashing)
378            .map_err(backend)?
379        else {
380            return Ok(None);
381        };
382        if !hashes_to(&hashing.hasher, &cid) {
383            return Err(covalence_lib_cid::Error::Mismatch.into());
384        }
385        Ok(Some(len))
386    }
