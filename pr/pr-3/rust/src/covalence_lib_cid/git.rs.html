1//! Git object IDs as CIDs. The block is a `git-raw` block: the inflated Git
2//! object `kind SP length NUL content` (kind blob, tree, commit or tag; length
3//! in minimal ASCII decimal). The digest is its SHA-1 or SHA-256 object ID.
4
5use sha2::Digest;
6
7use crate::{
8    Cid, Format, Hashed, Multicodec,
9    storage::{cid_header, fixed_header},
10};
11
12fixed_header!(
13    /// A Git object ID with SHA-1 (20 bytes), stored as the binary CID.
14    GitSha1,
15    4 + 20,
16    |codec, hash| codec == Multicodec::GIT_RAW && hash == Multicodec::SHA1
17);
18
19fixed_header!(
20    /// A Git object ID with SHA-256 (32 bytes), stored as the binary CID.
21    GitSha256,
22    4 + 32,
23    |codec, hash| codec == Multicodec::GIT_RAW && hash == Multicodec::SHA2_256
24);
25
26cid_header!(
27    /// A Git object ID with SHA-1 (20 bytes) or SHA-256 (32 bytes), stored as
28    /// the binary CID, zero-padded.
29    Git,
30    36,
31    |codec, hash, len| codec == Multicodec::GIT_RAW
32        && matches!((hash, len), (Multicodec::SHA1, 20) | (Multicodec::SHA2_256, 32))
33);
34
35impl Hashed for GitSha1 {
36    const FORMAT: Format = Format::GIT_SHA1;
37
38    fn digest_of(object: &[u8]) -> [u8; 20] {
39        sha1::Sha1::digest(object).into()
40    }
41}
42
43impl Hashed for GitSha256 {
44    const FORMAT: Format = Format::GIT_SHA256;
45
46    fn digest_of(object: &[u8]) -> [u8; 32] {
47        sha2::Sha256::digest(object).into()
48    }
49}

51/// Git object IDs, computed from the object with its header.
52///
53/// ```
54/// use covalence_lib_cid::{Cid, Git};
55///
56/// // The ID Git gives a file holding "hello\n".
57/// let id = Cid::<Git>::sha1(b"blob 6\0hello\n");
58/// assert_eq!(id.to_string(), "baf4bcfgoae3ckaylvdn2sbxxk2lh7hu4uokemsq");
59/// let hex: String = id.digest().iter().map(|b| format!("{b:02x}")).collect();
60/// assert_eq!(hex, "ce013625030ba8dba906f756967f9e9ca394464a");
61/// ```
62impl Cid<Git> {
63    /// The SHA-1 ID of `object`, a Git object with its header.
64    pub fn sha1(object: &[u8]) -> Self {
65        Cid::<GitSha1>::of(object).into()
66    }
67
68    /// The SHA-256 ID of `object`, a Git object with its header.
69    pub fn sha256(object: &[u8]) -> Self {
70        Cid::<GitSha256>::of(object).into()
71    }
