1//! The key-value store trait.
2//!
3//! A key-value (KV) store maps byte-string keys to byte values, and a put
4//! replaces any earlier value. [`Kv`] is the trait every store implements.
5//!
6//! - Sync methods `get`, `put` and `exists`, and async twins named as in
7//!   Python: the sync name prefixed with `a`, so `aget`, `aput` and `aexists`.
8//! - `Kv` is object safe: every method returns the shared [`Error`], and async
9//!   ones a boxed [`KvFuture`]. Hold a store as `Arc<dyn Kv>` (or take
10//!   `&dyn Kv`) rather than being generic; [`Kv::into_dyn`] makes one.
11//! - [`Kv::size`] and [`Kv::get_range`] fetch the whole value and measure or
12//!   slice it by default; stores with native ranges override them.
13//! - [`Kv::put_from`] and [`Kv::get_to`] move a value between the store and a
14//!   reader or writer. By default they hold it in memory; stores that can
15//!   stream override them. They have no async twins: readers and writers block.
16//! - [`Kv::put_new`], [`Kv::put_new_from`] and [`Kv::aput_new`] store a value
17//!   atomically only under a key that holds none. Their defaults return
18//!   [`Error::Unsupported`] without reading or writing; stores that can create
19//!   atomically override them.
20//! - Stores live in their own crates, such as `covalence-kv-memory` and
21//!   `covalence-kv-s3`.
22//!
23//! A store needs only `get`, `put`, `aget` and `aput`; the defaults supply
24//! the rest:
25//!
26//! ```
27//! use std::{collections::BTreeMap, sync::RwLock};
28//!
29//! use covalence_kv_trait::{Error, Kv, KvFuture, ready};
30//! use covalence_lib_data::Bytes;
31//!
32//! #[derive(Default)]
33//! struct Map(RwLock<BTreeMap<Vec<u8>, Bytes>>);
34//!
35//! impl Kv for Map {
36//!     fn get(&self, key: &[u8]) -> Result<Option<Bytes>, Error> {
37//!         Ok(self.0.read().unwrap().get(key).cloned())
38//!     }
39//!     fn put(&self, key: &[u8], value: Bytes) -> Result<(), Error> {
40//!         self.0.write().unwrap().insert(key.to_vec(), value);
41//!         Ok(())
42//!     }
43//!     fn aget<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, Option<Bytes>> {
44//!         ready(self.get(key))
45//!     }
46//!     fn aput<'a>(&'a self, key: &'a [u8], value: Bytes) -> KvFuture<'a, ()> {
47//!         ready(self.put(key, value))
48//!     }
49//! }
50//!
51//! let store = Map::default().into_dyn();
52//! store.put(b"greeting", "hello".into())?;
53//! assert_eq!(store.size(b"greeting")?, Some(5));
54//! assert_eq!(store.get_range(b"greeting", 1, 3)?.unwrap(), "ell");
55//! // Ranges stop at the end of the value; a missing key is `None`.
56//! assert_eq!(store.get_range(b"greeting", 3, 10)?.unwrap(), "lo");
57//! assert_eq!(store.get_range(b"greeting", 9, 1)?.unwrap(), "");
58//! assert_eq!(store.get_range(b"missing", 0, 1)?, None);
59//! assert!(!store.exists(b"missing")?);
60//! # Ok::<(), Error>(())
61//! ```
62#![deny(missing_docs)]

74/// The future an async store method returns.
75pub type KvFuture<'a, T> = Pin<Box<dyn Future<Output = Result<T, Error>> + Send + 'a>>;
76
77/// A [`KvFuture`] already holding `result`.
78pub fn ready<'a, T: Send + 'a>(result: Result<T, Error>) -> KvFuture<'a, T> {
79    Box::pin(std::future::ready(result))
80}

82/// A store of byte values by byte-string key, sync and async.
83///
84/// A store rejects keys it cannot hold with an error. Sync and async methods
85/// may see different data (sync might read only a local cache). Do not call
86/// the sync methods on an async runtime thread: they may block.
87pub trait Kv: Send + Sync {
88    /// The value under `key`, or `None` if missing.
89    fn get(&self, key: &[u8]) -> Result<Option<Bytes>, Error>;
90
91    /// The size in bytes of the value under `key`, or `None` if missing.
92    /// The length of [`Kv::get`] by default.
93    fn size(&self, key: &[u8]) -> Result<Option<u64>, Error> {
94        Ok(self.get(key)?.map(|value| value.len() as u64))
95    }
96
97    /// Up to `len` bytes under `key` from `start`, or `None` if missing.
98    /// Slices [`Kv::get`] by default.
99    ///
100    /// - Fewer bytes at the end, and empty bytes at or past it.
101    /// - Range ends beyond `u64::MAX` are clipped; replies never exceed `len`.
102    fn get_range(&self, key: &[u8], start: u64, len: u64) -> Result<Option<Bytes>, Error> {
103        Ok(self.get(key)?.map(|value| slice(&value, start, len)))
104    }
105
106    /// Store `value` under `key`, replacing any previous value.
107    fn put(&self, key: &[u8], value: Bytes) -> Result<(), Error>;
108
109    /// Store the next `len` bytes of `reader` under `key`, replacing any
110    /// previous value. Reads no further; fails with [`Error::Io`] if `reader`
111    /// fails or ends early, and then stores nothing. Reads them into memory
112    /// and calls [`Kv::put`] by default.
113    fn put_from(&self, key: &[u8], reader: &mut dyn Read, len: u64) -> Result<(), Error> {
114        let mut value = Vec::new();
115        let read = reader
116            .take(len)
117            .read_to_end(&mut value)
118            .map_err(|source| Error::Io { source })?;
119        if (read as u64) < len {
120            return Err(Error::Io {
121                source: io::ErrorKind::UnexpectedEof.into(),
122            });
123        }
124        self.put(key, value.into())
125    }
126
127    /// Atomically store `value` under `key` only if the key holds no value,
128    /// and return whether it stored.
129    ///
130    /// Never replaces a value, including when writers race. The default
131    /// returns [`Error::Unsupported`] without reading or writing.
132    fn put_new(&self, _key: &[u8], _value: Bytes) -> Result<bool, Error> {
133        Err(Op::PutNew.unsupported())
134    }
135
136    /// As [`Kv::put_from`], but atomically store only if the key holds no
137    /// value, and return whether it stored.
138    ///
139    /// Never replaces a value, including when writers race. When the key
140    /// holds one, `reader` may have been read or not. The default returns
141    /// [`Error::Unsupported`] without reading or writing.
142    fn put_new_from(&self, _key: &[u8], _reader: &mut dyn Read, _len: u64) -> Result<bool, Error> {
143        Err(Op::PutNew.unsupported())
144    }
145
146    /// Write the value under `key` to `writer` and return its length, or
147    /// `None`, writing nothing, if missing. Fails with [`Error::Io`] if
148    /// `writer` fails; after any failure `writer` may hold part of the value.
149    /// Writes [`Kv::get`] by default.
150    fn get_to(&self, key: &[u8], writer: &mut dyn Write) -> Result<Option<u64>, Error> {
151        let Some(value) = self.get(key)? else {
152            return Ok(None);
153        };
154        writer
155            .write_all(&value)
156            .map_err(|source| Error::Io { source })?;
157        Ok(Some(value.len() as u64))
158    }
159
160    /// The value under `key`, or `None` if missing.
161    fn aget<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, Option<Bytes>>;
162
163    /// As [`Kv::size`]. The length of [`Kv::aget`] by default.
164    fn asize<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, Option<u64>> {
165        Box::pin(async move { Ok(self.aget(key).await?.map(|value| value.len() as u64)) })
166    }
167
168    /// As [`Kv::get_range`]. Slices [`Kv::aget`] by default.
169    fn aget_range<'a>(
170        &'a self,
171        key: &'a [u8],
172        start: u64,
173        len: u64,
174    ) -> KvFuture<'a, Option<Bytes>> {
175        Box::pin(async move { Ok(self.aget(key).await?.map(|value| slice(&value, start, len))) })
176    }
177
178    /// Store `value` under `key`, replacing any previous value.
179    fn aput<'a>(&'a self, key: &'a [u8], value: Bytes) -> KvFuture<'a, ()>;
180
181    /// As [`Kv::put_new`], returning [`Error::Unsupported`] without reading
182    /// or writing by default.
183    fn aput_new<'a>(&'a self, _key: &'a [u8], _value: Bytes) -> KvFuture<'a, bool> {
184        ready(Err(Op::PutNew.unsupported()))
185    }
186
187    /// Whether there is a value under `key`. [`Kv::get`] by default; stores
188    /// override it to avoid reading the value.
189    fn exists(&self, key: &[u8]) -> Result<bool, Error> {
190        Ok(self.get(key)?.is_some())
191    }
192
193    /// Whether there is a value under `key`. [`Kv::aget`] by default; stores
194    /// override it to avoid reading the value.
195    fn aexists<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, bool> {
196        Box::pin(async move { Ok(self.aget(key).await?.is_some()) })
197    }
198
199    /// A DASL value describing the store, for debugging: `null` or a map.
200    /// `null` by default; an error if the store cannot answer.
201    fn info(&self) -> Result<Value, Error> {
202        Ok(Value::Null)
203    }
204
205    /// A DASL value of whatever the store counts, for debugging: `null` or a
206    /// map. `null` by default; an error if the store cannot answer.
207    fn stats(&self) -> Result<Value, Error> {
208        Ok(Value::Null)
209    }
210
211    /// This store as an `Arc<dyn Kv>`: wrapped in a new `Arc`, except that an
212    /// `Arc<dyn Kv>` returns itself.
213    fn into_dyn(self) -> Arc<dyn Kv>
214    where
215        Self: Sized + 'static,
216    {
217        Arc::new(self)
218    }
219}

221/// Up to `len` bytes of `value` from `start`, clipped at the end.
222fn slice(value: &Bytes, start: u64, len: u64) -> Bytes {
223    let size = value.len() as u64;
224    let first = start.min(size) as usize;
225    let last = start.saturating_add(len).min(size) as usize;
226    value.slice(first..last)
227}

229/// A shared store is a store: each method forwards to it.
230impl Kv for Arc<dyn Kv> {
231    fn get(&self, key: &[u8]) -> Result<Option<Bytes>, Error> {
232        (**self).get(key)
233    }
234    fn size(&self, key: &[u8]) -> Result<Option<u64>, Error> {
235        (**self).size(key)
236    }
237    fn get_range(&self, key: &[u8], start: u64, len: u64) -> Result<Option<Bytes>, Error> {
238        (**self).get_range(key, start, len)
239    }
240    fn put(&self, key: &[u8], value: Bytes) -> Result<(), Error> {
241        (**self).put(key, value)
242    }
243    fn put_from(&self, key: &[u8], reader: &mut dyn Read, len: u64) -> Result<(), Error> {
244        (**self).put_from(key, reader, len)
245    }
246    fn put_new(&self, key: &[u8], value: Bytes) -> Result<bool, Error> {
247        (**self).put_new(key, value)
248    }
249    fn put_new_from(&self, key: &[u8], reader: &mut dyn Read, len: u64) -> Result<bool, Error> {
250        (**self).put_new_from(key, reader, len)
251    }
252    fn get_to(&self, key: &[u8], writer: &mut dyn Write) -> Result<Option<u64>, Error> {
253        (**self).get_to(key, writer)
254    }
255    fn aget<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, Option<Bytes>> {
256        (**self).aget(key)
257    }
258    fn asize<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, Option<u64>> {
259        (**self).asize(key)
260    }
261    fn aget_range<'a>(
262        &'a self,
263        key: &'a [u8],
264        start: u64,
265        len: u64,
266    ) -> KvFuture<'a, Option<Bytes>> {
267        (**self).aget_range(key, start, len)
268    }
269    fn aput<'a>(&'a self, key: &'a [u8], value: Bytes) -> KvFuture<'a, ()> {
270        (**self).aput(key, value)
271    }
272    fn aput_new<'a>(&'a self, key: &'a [u8], value: Bytes) -> KvFuture<'a, bool> {
273        (**self).aput_new(key, value)
274    }
275    fn exists(&self, key: &[u8]) -> Result<bool, Error> {
276        (**self).exists(key)
277    }
278    fn aexists<'a>(&'a self, key: &'a [u8]) -> KvFuture<'a, bool> {
279        (**self).aexists(key)
280    }
281    fn info(&self) -> Result<Value, Error> {
282        (**self).info()
283    }
284    fn stats(&self) -> Result<Value, Error> {
285        (**self).stats()
286    }
287    fn into_dyn(self) -> Arc<dyn Kv> {
288        self
289    }
290}

292/// A store operation, named in [`Error::Unsupported`].
293#[derive(Clone, Copy, Debug, PartialEq, Eq)]
294#[non_exhaustive]
295pub enum Op {
296    /// A read.
297    Get,
298    /// A write.
299    Put,
300    /// An atomic create-only write.
301    PutNew,
302}
303
304impl Op {
305    /// The error for a store that does not support this operation.
306    pub fn unsupported(self) -> Error {
307        Error::Unsupported { op: self }
308    }
309}

311/// Errors from reading and storing.
312#[derive(Debug, Snafu)]
313#[snafu(crate_root(covalence_lib_error::snafu))]
314#[non_exhaustive]
315pub enum Error {
316    /// The store does not support `op`.
317    #[snafu(display("unsupported: {op:?}"))]
318    Unsupported {
319        /// The operation.
320        op: Op,
321    },
322    /// A failure in the storage under a store, including a key it cannot hold.
323    #[snafu(display("backend: {source}"))]
324    Backend {
325        /// The store's own error.
326        source: BoxError,
327    },
328    /// The reader [`Kv::put_from`] reads or the writer [`Kv::get_to`] writes
329    /// failed, or the reader ended early.
330    #[snafu(display("{source}"))]
331    Io {
332        /// The I/O error.
333        source: io::Error,
334    },
