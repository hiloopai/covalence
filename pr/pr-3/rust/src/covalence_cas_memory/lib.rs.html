1//! An in-memory content-addressed store.
2//!
3//! [`MemoryCas`] keeps each block once, as a [`Fact`] for its raw BLAKE3 CID.
4//!
5//! - A CID with a 32-byte [BLAKE3](https://github.com/BLAKE3-team/BLAKE3-specs)
6//!   digest finds its bytes whatever its codec; other CIDs find nothing.
7//! - [`Put`] returns the raw BLAKE3 CID, and `put_fact` of a BLAKE3 fact
8//!   stores it without hashing again.
9//! - The async methods return ready futures.
10//!
11//! ```
12//! use covalence_cas_memory::MemoryCas;
13//! use covalence_cas_trait::{GetExt, Put};
14//! use covalence_lib_cid::{Cid, Format};
15//! let store = MemoryCas::default();
16//! let cid: Cid = store.put("hello".into())?;
17//! assert_eq!(store.get_fact(&cid)?.unwrap().bytes(), "hello");
18//! // The same digest under another codec finds the same bytes.
19//! let drisl: Cid = Cid::new(Format::DRISL_BLAKE3, cid.digest())?;
20//! assert_eq!(store.get_fact(&drisl)?.unwrap().claim(), &drisl);
21//! // A block is stored once.
22//! let _: Cid = store.put("hello".into())?;
23//! assert_eq!(store.len(), 1);
24//! # Ok::<(), covalence_cas_trait::Error>(())
25//! ```
26#![deny(missing_docs)]
27

35/// An in-memory store of BLAKE3 facts; see the [crate] docs.
36///
37/// [`Cas::info`] is `{"name": "memory"}`, and [`Cas::stats`] counts `blocks`
38/// and their `bytes`.
39///
40/// # Panics
41///
42/// Putting a new block panics if the store already holds 2³² blocks.
43#[derive(Default)]
44pub struct MemoryCas(RwLock<Table>);
45
46/// Facts in insertion order, indexed by key.
47#[derive(Default)]
48struct Table {
49    facts: Vec<Fact<Blake3>>,
50    index: HashTable<u32>,
51}
52
53/// A table hash for `key`: its digest is already uniform.
54fn hash(key: &Blake3) -> u64 {
55    u64::from_le_bytes(key.digest()[..8].try_into().unwrap())
56}
57
58impl Table {
59    fn find(&self, key: &Blake3) -> Option<&Fact<Blake3>> {
60        let facts = &self.facts;
61        let i = self
62            .index
63            .find(hash(key), |&i| facts[i as usize].claim() == key)?;
64        Some(&facts[*i as usize])
65    }
66
67    fn insert(&mut self, fact: Fact<Blake3>) {
68        if self.find(fact.claim()).is_some() {
69            return;
70        }
71        let Table { facts, index } = self;
72        let i = u32::try_from(facts.len()).expect("fewer than 2^32 blocks");
73        index.insert_unique(hash(fact.claim()), i, |&i| hash(facts[i as usize].claim()));
74        facts.push(fact);
75    }
76}
77
78impl MemoryCas {
79    /// The number of stored blocks.
80    pub fn len(&self) -> usize {
81        self.0.read().unwrap().facts.len()
82    }
83
84    /// Whether the store is empty.
85    pub fn is_empty(&self) -> bool {
86        self.len() == 0
87    }
88}

90impl Get<Blake3> for MemoryCas {
91    fn get_unchecked(&self, key: &Blake3) -> Result<Option<Bytes>, Error> {
92        Ok(self.0.read().unwrap().find(key).map(|f| f.bytes().clone()))
93    }
94
95    fn get_fact_unchecked(&self, key: &Blake3) -> Result<Option<Fact<Blake3>>, Error> {
96        Ok(self.0.read().unwrap().find(key).cloned())
97    }
98}
99
100impl Get for MemoryCas {
101    fn get_unchecked(&self, key: &Cid) -> Result<Option<Bytes>, Error> {
102        key.raw_blake3()
103            .map_or(Ok(None), |key| self.get_unchecked(&key))
104    }
105
106    fn get_fact_unchecked(&self, key: &Cid) -> Result<Option<Fact>, Error> {
107        let Some(key) = key.raw_blake3() else {
108            return Ok(None);
109        };
110        Ok(self.get_fact_unchecked(&key)?.map(Fact::widen))
111    }
112}
113
114impl AsyncGet for MemoryCas {
115    fn aget_unchecked<'a>(&'a self, key: &'a Cid) -> CasFuture<'a, Option<Bytes>> {
116        ready(self.get_unchecked(key))
117    }
118
119    fn aget_fact_unchecked<'a>(&'a self, key: &'a Cid) -> CasFuture<'a, Option<Fact>> {
120        ready(self.get_fact_unchecked(key))
121    }
122}
123
124impl Put<Blake3> for MemoryCas {
125    fn put(&self, bytes: Bytes) -> Result<Blake3, Error> {
126        self.put_fact(Fact::of(bytes))
127    }
128
129    /// Stores `fact` without hashing.
130    fn put_fact(&self, fact: Fact<Blake3>) -> Result<Blake3, Error> {
131        let key = *fact.claim();
132        self.0.write().unwrap().insert(fact);
133        Ok(key)
134    }
135}
136
137impl Put for MemoryCas {
138    fn put(&self, bytes: Bytes) -> Result<Cid, Error> {
139        Ok(Put::<Blake3>::put(self, bytes)?.into())
140    }
141
142    /// Stores a BLAKE3 `fact` without hashing; puts any other.
143    fn put_fact(&self, fact: Fact) -> Result<Cid, Error> {
144        match fact.raw_blake3() {
145            Some(fact) => Ok(self.put_fact(fact)?.into()),
146            None => self.put(fact.bytes().clone()),
147        }
148    }
149}
150
151impl AsyncPut for MemoryCas {
152    fn aput(&self, bytes: Bytes) -> CasFuture<'_, Cid> {
153        ready(self.put(bytes))
154    }
155
156    fn aput_fact(&self, fact: Fact) -> CasFuture<'_, Cid> {
157        ready(self.put_fact(fact))
158    }
159}
160
161impl Cas for MemoryCas {
162    fn info(&self) -> Result<Value, Error> {
163        Ok(map([("name", Value::String("memory".into()))]))
164    }
165
166    fn stats(&self) -> Result<Value, Error> {
167        let table = self.0.read().unwrap();
168        let bytes = table.facts.iter().map(|f| f.bytes().len()).sum::<usize>();
169        Ok(map([
170            ("blocks", Value::Integer(table.facts.len() as i64)),
171            ("bytes", Value::Integer(bytes as i64)),
172        ]))
173    }
